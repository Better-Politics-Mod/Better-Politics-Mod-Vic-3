bpm_activate_law_as_codification_level = {
	activate_law = $law$
	random_active_law = {
		limit = { type = $law$ }
		bpm_law_set_codification_level = {
			level = $level$
		}
	}
}

bpm_law_set_codification_level = {
	if = {
		limit = {
			has_game_started = yes
		}
		owner.var:bpm_ruling_ig = { save_scope_as = sponsor_ig }
	}
	else = {
		owner.ig:ig_petty_bourgeoisie = { save_scope_as = sponsor_ig }
	}

	every_scope_amendment = {
		limit = { bpm_amendment_is_codification = yes }
		remove_amendment = yes
	}

	if = {
		limit = {
			flag:traditional = flag:$level$
		}
		add_amendment = {
			type = amendment_type:bpm_amendment_traditional
			sponsor = scope:sponsor_ig
		}
	}
	else_if = {
		limit = {
			flag:edict = flag:$level$
		}
		add_amendment = {
			type = amendment_type:bpm_amendment_edict
			sponsor = scope:sponsor_ig
		}
	}
	else_if = {
		limit = {
			flag:constitutional = flag:$level$
		}
		add_amendment = {
			type = amendment_type:bpm_amendment_constitutional
			sponsor = scope:sponsor_ig
		}
	}
}
