	value = 0

		save_temporary_scope_as = sr
	}

			variable = bpm_active_actions
			limit = {

					target = scope:sr
				}
			}
			add = 1
		}
	}
}

# this = state
# highest level of any revolutionary cell in this state (0 if none), used by the revolt JE widget
bpm_revolt_map_max_cell_level = {
	value = 0
	if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 3 } }
		value = 3
	}
	else_if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 2 } }
		value = 2
	}
	else_if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 1 } }
		value = 1
	}
}
