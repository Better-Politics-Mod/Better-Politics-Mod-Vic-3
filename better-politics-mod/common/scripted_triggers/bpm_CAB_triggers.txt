bpm_char_can_be_regent = {
    exists = interest_group
    is_character_alive = yes
    is_ruler_of_own_country = no
    is_heir_of_own_country = no
    NOR = {
	    has_role_of_type = agitator
	    has_role = bpm_character_role_hog
        has_role = bpm_character_role_acting_hog
    }
}

bpm_char_can_be_acting_hog = {
    exists = interest_group
    is_character_alive = yes
    is_ruler_of_own_country = no
    is_heir_of_own_country = no
    NOR = {
        has_role_of_type = agitator
        has_role = bpm_character_role_hog
        has_role = bpm_character_role_regent
    }
}

bpm_char_can_be_hog = {
    is_character_alive = yes
    is_ruler_of_own_country = no
    is_heir_of_own_country = no
    NOR = {
        has_role_of_type = agitator
        has_role = bpm_character_role_acting_hog
        has_role = bpm_character_role_regent
    }
    OR = {
        is_adult = no
        interest_group ?= { is_in_government = yes }
    }
}

bpm_character_is_hos = {
    trigger_if = {
        limit = { owner = { bpm_has_gov_regency = yes } }
        var:bpm_regent ?= this
    }
    trigger_else = {
        ruler ?= this
    }
}

bpm_character_is_hog = {
    trigger_if = {
        limit = { owner = { bpm_has_acting_hog = yes } }
        var:bpm_acting_hog ?= this
    }
    trigger_else = {
        var:bpm_headofgov.leader ?= this
    }
}
