bpm_country_has_republican_tradition = {
    active_law:lawgroup_governance_principles = {
        has_amendment = amendment_type:bpm_amendment_republican_tradition
    }
}

bpm_country_is_new_world_republic = {
    is_in_geographic_region = geographic_region_new_world
    bpm_country_is_republic = yes
}

bpm_country_has_different_heritage_primary_cultures = {
    save_temporary_scope_as = bpm_htc_country
    any_primary_culture = {
        save_temporary_scope_as = bpm_htc_culture_a
        scope:bpm_htc_country = {
            any_primary_culture = {
                NOT = { this = scope:bpm_htc_culture_a }
                NOT = { shares_heritage_trait_with_culture = scope:bpm_htc_culture_a }
            }
        }
    }
}

bpm_country_has_law_entrenched = {
    any_in_list = {
        variable = bpm_country_entrenched_laws
        type = $law$
    }
}

bpm_country_has_law_or_variant_entrenched = {
    any_in_list = {
        variable = bpm_country_entrenched_laws
        "global_variable_map(bpm_lawparent|type)" = $law$
    }
}

bpm_country_has_republic_entrenched = {
    OR = {
        bpm_country_has_law_entrenched = { law = law_type:law_presidential_republic }
        bpm_country_has_law_entrenched = { law = law_type:law_parliamentary_republic }
        bpm_country_has_law_entrenched = { law = law_type:law_council_republic }
    }
}

bpm_country_has_artisan_eco_entrenched = {
    OR = {
        active_law:lawgroup_economic_system = {
            has_amendment = amendment_type:bpm_amendment_artisanal_economy
        }
        has_variable = bpm_recently_had_artisanal_economy
    }
}
