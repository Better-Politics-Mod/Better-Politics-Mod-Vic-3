bpm_journal_progress_complete = {
	custom_tooltip = {
		text = "bpm_je_progress_complete"
		var:$VAR$ >= 100
	}
}

bpm_journal_progress_zero = {
	custom_tooltip = {
		text = "bpm_je_progress_zero"
		has_variable = $VAR$
		var:$VAR$ <= 0
	}
}

bpm_has_law_change_journal = {
	OR = {
		has_journal_entry = je_government_petition  
		has_journal_entry = je_law_imposition
	}
}

bpm_modernizers_emulate_this_country = {
	is_country_type = recognized
	OR = {
		has_variable = bpm_je_modernizers_renamed_into_modernizers
		is_in_geographic_region = geographic_region_europe
		is_in_geographic_region = geographic_region_new_world
	}
	is_subject = no
}

bpm_springtime_over = {
	OR = {
		hidden_trigger = { 
			OR = {
				has_global_variable = peoples_springtime_happened
				game_date > 1865.1.1
			}
		}
		AND = {
			hidden_trigger = { has_game_started = yes }
			any_country_in_europe = {
				count = all
				NOR = {
					has_journal_entry = je_forest_of_bayonets
					has_journal_entry = je_springtime_of_the_peoples
					has_journal_entry = je_red_summer
				}
			}
		}
	}
}
