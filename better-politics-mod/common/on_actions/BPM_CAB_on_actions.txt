on_monthly_pulse_country = {
    on_actions = {
        bpm_reload_hog
    }
}

on_government_reformed = {
	on_actions = {
		bpm_reload_hog
	}
}

on_election_campaign_start = {
	on_actions = {
		bpm_reload_hog
	}
}

on_election_campaign_end = {
	on_actions = {
		bpm_reload_hog
	}
}

on_character_death = {
    on_actions = {
        bpm_update_chars
    }
}

bpm_update_chars = {
    effect = {
        if = {
            limit = {
                owner.var:bpm_headofgov ?= THIS
                has_variable = bpm_rightfully_elected_headofgov
                owner = {
                    bpm_country_has_parliamentary_electoral_system = yes
                }
            }
            owner = {
                trigger_event = {
                    id = bpm_form_government_events.2
                    days = 1
                    popup = yes
                }
            }
        }
        owner = {
            bpm_choose_hog = yes
            bpm_choose_regent = yes
            bpm_choose_acting_hog = yes
            bpm_update_hog_heir = yes
        }
    }
}

bpm_reload_hog = {
    effect = {
        bpm_choose_hog = yes
    }
}
