# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law

REPLACE:law_consumption_based_taxation = {
	group = lawgroup_taxation

	progressiveness = -100

	icon = "gfx/interface/icons/law_icons/consumption_based_taxation.dds"

	modifier = {
		state_bureaucracy_population_base_cost_factor_mult = -0.25
		country_consumption_tax_cost_mult = -0.50
	}

	tax_modifier_very_low = {
		tax_consumption_add = 0.3
	}

	tax_modifier_low = {
		tax_consumption_add = 0.4
	}

	tax_modifier_medium = {
		tax_consumption_add = 0.5
	}

	tax_modifier_high = {
		tax_consumption_add = 0.6
	}

	tax_modifier_very_high = {
		tax_consumption_add = 0.7
	}

	# AI shouldn't change to consumption-based-taxation as it's likely to wreck their economy
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

REPLACE:law_per_capita_based_taxation = {
	group = lawgroup_taxation

	progressiveness = 0

	icon = "gfx/interface/icons/law_icons/per_capita_based_taxation.dds"

	disallowing_laws = {
		law_traditionalism
	}

	unlocking_technologies = {
		currency_standards
	}

	tax_modifier_very_low = {
		tax_consumption_add = 0.3
		tax_per_capita_add = 0.2
		tax_land_add = 0.20
	}

	tax_modifier_low = {
		tax_consumption_add = 0.4
		tax_per_capita_add = 0.275
		tax_land_add = 0.275
	}

	tax_modifier_medium = {
		tax_consumption_add = 0.5
		tax_per_capita_add = 0.35
		tax_land_add = 0.35
	}

	tax_modifier_high = {
		tax_consumption_add = 0.6
		tax_per_capita_add = 0.425
		tax_land_add = 0.425
	}

	tax_modifier_very_high = {
		tax_consumption_add = 0.7
		tax_per_capita_add = 0.5
		tax_land_add = 0.50
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		# Once the AI gets to a certain society tech level they should try to ditch Land Taxation for Per-Capital Taxtion
		if = {
			limit = {
				OR = {
					has_technology_researched = civilizing_mission
					has_technology_researched = human_rights
					has_technology_researched = pan-nationalism
				}
				has_law_or_variant = law_type:law_land_based_taxation
			}

			add = 1000
		}

		if = {
			limit = {
				OR = {
					has_journal_entry = je_sick_man_bureaucracy
					has_journal_entry = je_meiji_main
					has_journal_entry = je_great_reforms_bureaucratic
					has_journal_entry = je_hawaii
					has_journal_entry = je_portugal_regeneration
				}
				has_law_or_variant = law_type:law_land_based_taxation
			}
			add = 5000
		}
	}
}
