bpm_ig_monthly_radicals_from_approval = {
    value = 0.005
    add = {
        value = 0.02
        multiply = {
            value = ig_approval
            divide = define:NPolitics|MIN_IG_APPROVAL
        }
        min = 0
    }
    multiply = {
        value = 1
        add = {
            value = modifier:country_radicals_from_low_approval_mult
            if = {
                limit = { bpm_ig_is_proletarian = yes }
                add = modifier:country_radicals_from_low_approval_ig_proletarian_mult
            }
        }
    }
}

bpm_ig_loyalists_from_enactment = {

                    law = owner.currently_enacting_law.type
