bpm_can_revolt_ig = {
	NOR = {
		has_modifier = bpm_adapting_to_new_regime
		has_modifier = bpm_recently_failed_revolt
	}
}

bpm_can_revolt_country = {
	any_scope_state = {
		# possible for revcell to spawn
		turmoil >= 0.25
	}
}

# this = ig
# scope:ig
bpm_ig_is_hostile = {
	bpm_inter_ig_approval < 0.5
}

bpm_ig_is_friendly = {
	bpm_inter_ig_approval >= 0.75
}

# this = action
# scope:ig
# If friendly action clout > 50%
# or friendly action clout > 25% and hostile action clout = 0%
# then the IG is willing to join the action
bpm_ig_willing_to_join_action = {
	save_temporary_scope_value_as = {
		name = bpm_friendly_clout
		value = bpm_friendly_action_clout
	}
	scope:bpm_friendly_clout > 0.25
	OR = {
		scope:bpm_friendly_clout > 0.5
		bpm_hostile_action_clout < 0.01
	}
}

# this = ig
# An angry IG never supports a coalition negotiation, whatever sways it received.
# The ruling IG itself is exempt, as it is the one forming the coalition.
bpm_ig_always_opposes_coalition = {
	ig_approval <= angry
	NOT = { owner.var:bpm_ruling_ig ?= this }
}
