ig_liberals = {
	color = hsv{ 0.16 1 1 }
	texture = "gfx/interface/icons/ig_icons/liberals.dds"
	layer = "revolution_dynamic_industrialists"
	index = 12
	
	ideologies = {
		ideology_nonexistent
		
		# Vanilla
		ideology_liberal
		ideology_laissez_faire
		ideology_individualist
	}
	
	character_ideologies = {
		ideology_center_leader
		ideology_protectionist
		ideology_land_reformer
	}

	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_liberals = yes
		bpm_update_name_liberals = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				assamese avadhi baluchi bengali bihari gujarati kannada malayalam marathi oriya panjabi pathan rajput sindi tamil telegu hindustani bundeli pahari gondi bageli chhattisgarhi deccani
			}
		}
	}

	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				ig:ig_liberals ?= {
					bpm_interest_group_is_activated = yes
				}
			}
		}
	}
	
	pop_weight = {
		value = 0

		if = {
			limit = {
				strata = upper
			}
			add = { 
				value = 35
				desc = "UPPER_CLASS"
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 15
					desc = "POP_CAPITALISTS"
				}
			}
		}
		else = {
			add = {
				desc = "SELF_OWNED_WORKPLACE"
				if = { 
					limit = {
						exists = workplace
					}
					value = 20
					multiply = { workplace ?= { value = self_ownership_fraction } }
				}
				if = {
					limit = {
						strata = middle
					}
					multiply = 1.5
				}
			}
		}

		add = {
			value = wealth 
			subtract = 10
			min = 0
			max = 50
		}

		add = {
			if = {
				limit = {
					OR = {
						pop_employment_building_group = bg_trade
						pop_employment_building = building_port
					}
					NOT = {
						strata = upper
					}
				}
				value = 15
			}
			desc = "POP_INTERNATIONAL_TRADE"
		}

		multiply = {
			desc = "MULTIPLIER"
			value = 1
			subtract = {
				value = 0.5
				subtract = literacy_rate
				min = 0
				divide = 5
				desc = "LOW_LITERACY"
			}
			if = {
				limit = {
					has_pop_religion = atheist
				}
				add = {
					add = 0.1
					desc = "ATHEIST"
				}
			}
			else_if = {
				limit = {
					NOT = {
						has_state_religion = no
					}
				}
				add = {
					value = 0.1
					desc = "RELIGIOUS_MINORITY"
				}
			}


		}
	}
	
	monarch_weight = {
		owner.ig:ig_liberals = {
			value = bpm_ig_monarch_weight
		}
	}
	
	agitator_weight = {
		owner.ig:ig_liberals = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_liberals = {
			value = bpm_commander_weight
		}
	}	

	executive_weight = {
		value = 0.5
		if ={
			limit = {
				owner.ig:ig_liberals = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
