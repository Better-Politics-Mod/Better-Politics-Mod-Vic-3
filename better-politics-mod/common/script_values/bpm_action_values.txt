# this = state
# highest level of any revolutionary cell in this state (0 if none), used by the revolt JE widget
bpm_revolt_map_max_cell_level = {
	value = 0
	if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 4 } }
		value = 4
	}
	else_if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 3 } }
		value = 3
	}
	else_if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 2 } }
		value = 2
	}
	else_if = {
		limit = { bpm_revolt_map_cell_level = { lvl = 1 } }
		value = 1
	}
}

# this = country
# 0-6, the grasp category the shared front line settles on: higher with the clout of all revolutions, lower with legitimacy
bpm_revolt_front_line_centre = {
	value = 3
	if = {
		limit = {
			bpm_revolt_total_clout > 0
		}
		value = bpm_revolt_total_clout
		divide = {
			value = bpm_revolt_total_clout
			add = legitimacy
		}
		multiply = 6
		round = yes
	}
	min = 0
	max = 6
}
