bpm_update_political_stability = {
	if = {
		limit = { NOT = { has_global_variable_list = bpm_rigidity_levels } }
		bpm_setup_rigidity_levels = yes
	}

	if = {
		limit = { NOT = { has_variable = bpm_political_stability } }
		set_variable = { name = bpm_political_stability value = bpm_political_stability_baseline_target }
		recalculate_pop_ig_support = yes
	} else = {
		bpm_handle_baseline_target_trend_script_value = {
			VARIABLE = bpm_political_stability
			RATE = bpm_political_stability_rate
		}
	}

	clamp_variable = {
		name = bpm_political_stability
		min = 0 max = 100
	}

	# HANDLING STATIC MODIFIERS ---------------------------------
	remove_modifier = bpm_stability_stagnant
	remove_modifier = bpm_stability_stagnant_ai
	remove_modifier = bpm_stability_entrenched
	remove_modifier = bpm_stability_entrenched_ai
	remove_modifier = bpm_stability_dynamic
	remove_modifier = bpm_stability_unstable
	remove_modifier = bpm_stability_unstable_buffed
	remove_modifier = bpm_stability_chaotic
	remove_modifier = bpm_stability_chaotic_buffed

	if = {
		limit = { bpm_rigidity_is_stagnant = yes }
		if = {
			limit = {
				OR = {
					has_game_started = no
					is_player = yes
				}
			}
			add_modifier = { name = bpm_stability_stagnant }
		} else = {
			add_modifier = { name = bpm_stability_stagnant_ai }
		}
	} else_if = {
		limit = { bpm_rigidity_is_entrenched = yes }
		if = {
			limit = {
				OR = {
					has_game_started = no
					is_player = yes
				}
			}
			add_modifier = { name = bpm_stability_entrenched }
		} else = {
			add_modifier = { name = bpm_stability_entrenched_ai }
		}
	} else_if = {
		limit = { bpm_rigidity_is_dynamic = yes }
		add_modifier = { name = bpm_stability_dynamic }
	} else_if = {
		limit = { bpm_rigidity_is_unstable = yes }
		if = {
			limit = { modifier:country_bpm_low_rigidity_no_econ_debuff = no }
			add_modifier = { name = bpm_stability_unstable }
		} else = {
			add_modifier = { name = bpm_stability_unstable_buffed }
		}
	} else = {
		if = {
			limit = { modifier:country_bpm_low_rigidity_no_econ_debuff = no }
			add_modifier = { name = bpm_stability_chaotic }
		} else = {
			add_modifier = { name = bpm_stability_chaotic_buffed }
		}
	}
}
