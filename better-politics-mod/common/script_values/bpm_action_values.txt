bpm_revolt_progress_value = {
	value = 0
	every_scope_state = {
		limit = {
			state_region = {
				bpm_has_revcell = yes
			}
		}
		add = bpm_revolt_progress_value_state
	}
}

bpm_revolt_progress_value_state = {
	value = 0
	if = {
		limit = {
			state_region = {
				bpm_has_revcell_level = { lvl = 1 }
			}
		}
		add = 1
	}
	else_if = {
		limit = {
			state_region = {
				bpm_has_revcell_level = { lvl = 2 }
			}
		}
		add = 3
	}
	else_if = {
		limit = {
			state_region = {
				bpm_has_revcell_level = { lvl = 3 }
			}
		}
		add = 10
	}
	multiply = bpm_population_fraction
}

bpm_population_fraction = {
	value = state_population
	divide = owner.total_population
}
