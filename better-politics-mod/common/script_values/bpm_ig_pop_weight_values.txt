bpm_pop_weight_inner_ig_armed_forces = {
	value = 0

	if = {
		limit = { is_pop_type = soldiers }
		add = {
			desc = "POP_SOLDIERS"
			value = scope:interest_group.modifier:interest_group_soldiers_attraction_add
		}
	}
	else_if = {
		limit = { is_pop_type = aristocrats }
		add = {
			desc = "POP_ARISTOCRATS"
			value = 5
			add = state.modifier:state_aristocrats_ig_armed_forces_attraction_add
		}
	}
	else_if = {
		limit = { is_pop_type = officers }
		add = {
			desc = "POP_OFFICERS"
			value = 50
		}
	}
	else_if = {
		limit = { is_pop_type = bureaucrats }
		add = {
			desc = "POP_BUREAUCRATS"
			value = 5
		}
	}
	
	if = {
		limit = { pop_acceptance >= acceptance_status_4 }
		add = {
			desc = "POP_NOT_DISCRIMINATED"
			value = 25
			if = { # Militarisation of colonial settlers
				limit = {
					state = {
						is_incorporated = no
					}
				}
				add = {
					desc = "state_unincorporated_ig_armed_forces_attraction_add"
					value = state.modifier:state_unincorporated_ig_armed_forces_attraction_add
				}
			}
			if = { # Separation of the military and people
				limit = {
					NOR = {
						is_pop_type = soldiers
						is_pop_type = officers
					}
				}
				multiply = {
					desc = "state_non_discriminated_civilian_ig_armed_forces_attraction_mult"
					value = 1
					add = state.modifier:state_non_discriminated_civilian_ig_armed_forces_attraction_mult
					min = 0
				}
			}
		}
	}	

	multiply = { value = ig_attraction_to_interests_from_stability desc = "BPM_POP_INTERESTS_FROM_STABILITY" }
	multiply = { value = bpm_pop_weight_multiplier_ig_armed_forces desc = "MULTIPLIER" }
}
