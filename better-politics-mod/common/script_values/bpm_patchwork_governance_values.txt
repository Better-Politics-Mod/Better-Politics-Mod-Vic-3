# this = state
bpm_patchwork_governance_level = {
	value = 0
	if = {
		limit = { has_variable = bpm_patchwork_governance_level }
		add = var:bpm_patchwork_governance_level
	}
}

# this = state
bpm_patchwork_governance_progress = {
	value = 0
	if = {
		limit = { has_variable = bpm_patchwork_governance_progress }
		add = var:bpm_patchwork_governance_progress
	}
}

# this = state
bpm_patchwork_governance_speed_mult = {
	value = 1
	if = {
		limit = { is_incorporated = yes }
		if = {
			limit = { exists = owner.ig:ig_petty_bourgeoisie }
			add = {
				value = owner.ig:ig_petty_bourgeoisie.ig_clout
				multiply = owner.ig:ig_petty_bourgeoisie.ig_approval
				divide = 10
				desc = "bpm_patchwork_governance_speed_from_petty_bourgeoisie"
			}
		}
	}
	else = {
		if = {
			limit = { exists = owner.ig:ig_armed_forces }
			add = {
				value = owner.ig:ig_armed_forces.ig_clout
				multiply = owner.ig:ig_armed_forces.ig_approval
				divide = 10
				desc = "bpm_patchwork_governance_speed_from_armed_forces"
			}
		}
	}
	if = {
		limit = { exists = owner.ig:ig_landowners }
		subtract = {
			value = owner.ig:ig_landowners.ig_clout
			multiply = owner.ig:ig_landowners.ig_approval
			divide = 10
			desc = "bpm_patchwork_governance_speed_from_landowners"
		}
	}
	min = 0.25
	max = 2
}

# this = state
bpm_patchwork_governance_monthly_progress = {
	value = 100
	divide = 24
	multiply = bpm_patchwork_governance_speed_mult
}
