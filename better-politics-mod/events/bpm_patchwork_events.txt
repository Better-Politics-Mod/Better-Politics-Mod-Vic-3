namespace = bpm_patchwork_events

bpm_patchwork_events.1 = {
	type = country_event
	placement = root

    dlc = dlc_bpm

	duration = 3

	title = bpm_patchwork_events.1.t
	desc = {
        first_valid = {
            triggered_desc = {
                desc = bpm_patchwork_events.1.d
                trigger = { NOT = { exists = scope:too_few_states } }
            }
            triggered_desc = {
                desc = bpm_patchwork_events.1.d_alt
                trigger = { exists = scope:too_few_states }
            }
        }
    }
    flavor = bpm_patchwork_events.1.f

    event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    immediate = {
        capital = {
            save_scope_as = capital_state
        }
        ordered_scope_state = {
            limit = {
                is_incorporated = yes
                is_capital = no
                is_split_state = no
            }
            alternative_limit = {
                always = yes
            }
            order_by = state_population
            position = 0
            save_scope_as = state_1
        }
        ordered_scope_state = {
            limit = {
                is_incorporated = yes
                is_capital = no
                is_split_state = no
                NOT = { scope:state_1 = this }
            }
            alternative_limit = {
                always = yes
            }
            order_by = { subtract = state_population }
            position = 0
            save_scope_as = state_2
        }
        if = {
            limit = {
                OR = {
                    scope:state_1 = scope:state_2
                    scope:state_1 = scope:capital_state
                    scope:state_2 = scope:capital_state
                }
            }
            save_scope_value_as = {
                name = too_few_states
                value = yes
            }
        }
    }

	option = {
		name = bpm_patchwork_events.1.a
		default_option = yes
        active_law:lawgroup_centralization = { bpm_amendment_remove_util = { type = bpm_amendment_patchwork_governance } }
	}
}
