je_bpm_italian_war_of_independence = {
	group = je_group_bpm_crises
	icon = "gfx/interface/icons/event_icons/italy.dds"

	modifiers_while_active = {
		bpm_struggle_for_lombardy_venetia
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_bpm_crises_italian_war_of_independence
			}
		}
	}

	#scripted_progress_bar = je_bpm_italian_war_of_independence_progress_bar

	can_deactivate = yes

	immediate = {
		set_global_variable = {
			name = bpm_italian_independence_revolutionaries_score
			value = 0
		}
		set_global_variable = {
			name = bpm_italian_independence_austrians_score
			value = 0
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_VENETIA
			}
			save_scope_as = 1848_san_marco_source
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_LOMBARDY
			}
			save_scope_as = 1848_5_days_of_milan
		}
		c:SAR = {
			save_scope_as = 1848_sardinia_piedmont_country
		}
		c:AUS = {
			save_scope_as = 1848_austria_empire_country
		}
	}


	complete = {
		custom_tooltip = {
			text = bpm_one_side_has_won_10_battles
			OR = {
				global_var:bpm_italian_independence_revolutionaries_score >= 5
				global_var:bpm_italian_independence_austrians_score >= 5
			}
		}
	}

	on_complete = {
		if = {
			limit = { # If Austrians had won 5 battles
				global_var:bpm_italian_independence_austrians_score >= 5
			}
			c:SAR = {
				trigger_event = {
					id = bpm_springtime_of_the_peoples.13
					popup = yes
				}
			}
		}
		else_if = {
			limit = { # If the Italians had won 5 battles
				global_var:bpm_italian_independence_revolutionaries_score >= 5
			}
			random_diplomatic_play = {
				limit = {
					any_scope_play_involved = {
						c:LOM ?= this
					}
				}
				end_play = yes
			}
		}
	}

	should_be_pinned_by_default = no
}
