bpm_can_revolt_ig = {
	NOR = {
		has_modifier = bpm_adapting_to_new_regime
		has_modifier = bpm_recently_failed_revolt
	}
}

bpm_can_revolt_country = {
	any_scope_state = {
		# possible for revcell to spawn
		turmoil >= 0.25
	}
}

# this = ig
# scope:ig
bpm_ig_is_hostile = {
	bpm_inter_ig_approval < 0.5
}

bpm_ig_is_friendly = {
	bpm_inter_ig_approval >= 0.75
}

# this = action
# scope:ig
# If friendly action clout > 50%
# or friendly action clout > 25% and hostile action clout = 0%
# then the IG is willing to join the action
bpm_ig_willing_to_join_action = {
	OR = {
		bpm_friendly_action_clout > 0.5
		AND = {
			bpm_friendly_action_clout > 0.25
			bpm_hostile_action_clout < 0.01
		}
	}
}
