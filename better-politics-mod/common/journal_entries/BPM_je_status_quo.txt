je_bpm_status_quo_armed_neutrality = {
	icon = "gfx/interface/icons/invention_icons/trench_works.dds"

	group = je_group_internal_affairs

	immediate = {
		set_variable = {
			name = bpm_armed_neutrality_entrench_var
			value = 0
		}
	}

	is_progressing = {
		has_law = law_type:law_armed_neutrality
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = "bpm_je_progress_complete"
				has_variable = bpm_armed_neutrality_entrench_var
				var:bpm_armed_neutrality_entrench_var >= 10
			}
			has_law = law_type:law_armed_neutrality
		}
		effect = {
			every_interest_group = {
				limit = {
					OR = {
						has_ideology = ideology:ideology_mil_patriotic
						has_ideology = ideology:ideology_mil_patriotic_liberal
					}
				}
				bpm_remove_all_mil_ideologies = yes
				add_ideology = ideology_mil_defensive_patriot 
				bpm_reorder_ideologies = yes
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_interest_group = {
					OR = {
						has_ideology = ideology:ideology_mil_patriotic
						has_ideology = ideology:ideology_mil_patriotic_liberal
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = "bpm_je_progress_zero"
				has_variable = bpm_armed_neutrality_entrench_var
				var:bpm_armed_neutrality_entrench_var <= 0
			}
			NOT = { has_law = law_type:law_armed_neutrality }
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						is_progressing = yes
					}
				}
				change_variable = {
					name = bpm_armed_neutrality_entrench_var
					add = 1
				}
				clamp_variable = {
					name = bpm_armed_neutrality_entrench_var
					max = 10
					min = 0
				}
			} else = {
				change_variable = {
					name = bpm_armed_neutrality_entrench_var
					subtract = 1
				}
				clamp_variable = {
					name = bpm_armed_neutrality_entrench_var
					max = 10
					min = 0
				}
			}
		}
	}

	current_value = {
		value = var:bpm_armed_neutrality_entrench_var
	}

	goal_add_value = {
		value = 10
	}

	progressbar = yes

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = no
}
