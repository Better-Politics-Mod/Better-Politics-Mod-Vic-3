ig_national_liberals = {
	color = hsv{ 0.50 0.9 0.8 }
	texture = "gfx/interface/icons/ig_icons/national_liberals.dds"
	layer = "revolution_dynamic_petty_bourgeoisie"
	index = 10

	ideologies = {
		ideology_nonexistent
	}
	
	character_ideologies = {
		ideology_center_leader
		ideology_jingoist_leader
		ideology_protectionist
		ideology_sovereignist_leader
		ideology_corporatist_leader
	}

	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_national_liberals = yes
		bpm_update_name_national_liberals = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				assamese avadhi baluchi bengali bihari gujarati kannada malayalam marathi oriya panjabi pathan rajput sindi tamil telegu hindustani bundeli pahari gondi bageli chhattisgarhi deccani
			}
		}

		rule = {
			trigger = {
				c:CHI ?= this
				any_primary_culture = { 
					cu:manchu = this
				}
			}
			cultures = {
				han
			}
		}
	}


	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				ig:ig_national_liberals ?= {
					bpm_interest_group_is_activated = yes
				}
			}
		}
	}

	pop_weight = {
		value = 0

		# Main intended support

		if = {
			limit = { 
				OR = {
					is_pop_type = capitalists
					is_pop_type = shopkeepers
				} 
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					desc = "POP_CAPITALISTS"
					value = 50
				}
			}
			else_if = {
				limit = {
					OR = {
						pop_employment_building_group = bg_extraction
						pop_employment_building = building_tooling_workshop
						pop_employment_building = building_steel_mill
					}
				}
				add = { 
					# loc here
					value = 20
				}
			}
			else = {
				add = {
					desc = "POP_SHOPKEEPERS"
					value = 10
				}
			}
			add = {
				desc = "LIBERAL_CONSCIOUSNESS"
				value = owner.bpm_consciousness_liberal_igs_capitalists_shopkeepers
			}
		}

		else_if = {
			limit = {
				is_pop_type = academics
			}
			add = {
				desc = "POP_ACADEMICS"
				value = 20
			}
			add = {
				desc = "LIBERAL_CONSCIOUSNESS"
				value = owner.bpm_consciousness_liberal_igs_academics
			}
		}

		else_if = {
			limit = {
				is_pop_type = engineers
			}
			if = {
				limit = {
					OR = {
						pop_employment_building_group = bg_extraction
						pop_employment_building = building_tooling_workshop
						pop_employment_building = building_steel_mill
					}
				}
				add = {
					# loc here
					value = 20
				}
			}
			else = {
				add = {
					desc = "POP_ENGINEERS"
					value = 10
				}
			}
			add = {
				desc = "LIBERAL_CONSCIOUSNESS" 
				value = owner.bpm_consciousness_liberal_igs_engineers
			}
		}

		# General support

		else_if = {
			limit = {
				strata = middle
				NOT = { is_pop_type = clergymen }
			}
			add = {
				desc = "MIDDLE_CLASS"
				value = 20
			}
		}

		else = {
			add = {
				desc = "LIBERAL_CONSCIOUSNESS"
				value = owner.bpm_consciousness_liberal_igs_general 
				multiply = 1.2
				round = yes
			} # Should also add support of the national movement if present
		}

		add = {
			desc = "EDUCATED"
			value = literacy_rate
			subtract = owner.literacy_rate
			multiply = 100
			max = 50
		}

		# Flavour

		if = { 
			limit = { 
				owner = {
					has_global_variable = bpm_peoples_springtime_happened
					OR = {
						can_form_nation = ITA
						can_form_nation = GER
					}
				}
			}
			add = { 
				desc = "UNIFICATION_CANDIDATE"
				value = 10
			}
		}
		else_if = { 
			limit = { 
				owner = { 
					has_modifier = unification_prestige
				}
			}
			add = { 
				desc = "RECENTLY_UNIFIED"
				value = 50
			}
		}

		multiply = {
			desc = "IN_WAR"	
			value = 1.0
			if = {
				limit = { owner = { is_at_war = yes } }
				value = 1.5
			}		
		}

		multiply = ig_attraction_from_leader_popularity
		multiply = bpm_ig_agitators_popularity
		multiply = bpm_ig_ruler_popularity
		multiply = { desc = bpm_ig_attraction_from_radicals_and_loyalists_tt value = ig_attraction_from_radicals_and_loyalists }
		multiply = ig_attraction_to_ruler_ideology_from_stability
		multiply = ig_attraction_from_government
	}
	
	monarch_weight = {
		owner.ig:ig_national_liberals = {
			value = bpm_ig_monarch_weight
		}
	}
	
	agitator_weight = {
		owner.ig:ig_national_liberals = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_national_liberals = {
			value = bpm_commander_weight
		}
	}

	executive_weight = {
		value = 0.3
		if ={
			limit = {
				owner.ig:ig_national_liberals = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
