bpm_lawgroup_size = {
	value = 0
	save_temporary_scope_as = law
	owner = {
		every_law = {
			limit = {
				type = { save_temporary_scope_as = law_type }
				scope:law_type = {
					is_same_law_group_as = scope:law.type
				}
			}
			add = 1
		}
	}
}

bpm_aggregate_salience = {
	value = bpm_salience_weight_of_igs
	divide = {
		owner = { 
			every_active_law = {
				add = bpm_salience_weight_of_igs
			}
			min = 1
		}
	}
}

bpm_salience_weight_of_ig = {
	value = "variable_map(bpm_ig_salience_weights|scope:law.type)"
}

bpm_salience_weight_of_igs = {
	save_temporary_scope_as = law
	scope:first_ig = {
		add = bpm_salience_weight_of_ig
	}
	scope:second_ig = {
		add = bpm_salience_weight_of_ig
	}
	if = {
		limit = {
			OR = {
				scope:first_ig.bpm_salience_weight_of_ig = 0
				scope:second_ig.bpm_salience_weight_of_ig = 0
			}
		}
		multiply = 0
	}
}

bpm_salience_weight = {
	value = 1
	save_temporary_scope_as = law
	if = {
		limit = {
			type = { save_temporary_scope_as = law_type }
			scope:law_type = {
				OR = {
					is_same_law_group_as = law_type:law_monarchy
					is_same_law_group_as = law_type:law_autocracy
				}
			} 
		}
		multiply = 3
	}
	else_if = {
		limit = {
			type = { save_temporary_scope_as = law_type }
			scope:law_type = {
				OR = {
					is_same_law_group_as = law_type:law_repeatable_politics_none
					is_same_law_group_as = law_type:law_repeatable_culture_none
					is_same_law_group_as = law_type:law_repeatable_economy_none
					is_same_law_group_as = law_type:law_repeatable_military_none
					is_same_law_group_as = law_type:law_repeatable_religion_none
					is_same_law_group_as = law_type:law_no_separate_hog
					is_same_law_group_as = law_type:law_professional_navy
					AND = { 
						prev.owner = { has_law = law_type:law_slavery_banned }
						is_same_law_group_as = law_type:law_slavery_banned
					}
				} 
			}
		}
		multiply = 0
	}
	else = { 
		multiply = 1
	}

	if = {
		limit = {
			AND = {
				scope:ig = { is_interest_group_type = ig_trade_unions }
				scope:law_type = { is_same_law_group_as = law_type:law_combination_acts }
			}
		}
		multiply = 5
	}

	if = {
		limit = {
			OR = {
				scope:ig = {
					has_ideology = "global_variable_map(bpm_law_ideology_group|scope:law.type)"
				}
				AND = {
					scope:law_type = {
						is_same_law_group_as = law_type:law_autocracy
					}
					scope:ig = {
						OR = {
							has_ideology = ideology:ideology_dop_unionist
							has_ideology = ideology:ideology_dop_state_unionist
						}
					}
				}
			}
		}
		multiply = 0
	}
	# if = {

	# 	}
	# 	multiply = owner.bpm_consensus_on_lawgroup

bpm_inter_ig_approval = {
	value = 1
	save_temporary_scope_as = first_ig
	scope:ig = {
		save_temporary_scope_as = second_ig
	}
	owner = {
		every_law = {
			limit = {
				owner = {
					NOT = { has_law = prev.type }
				}
			}
			save_temporary_scope_as = law
			subtract = {
				value = scope:first_ig.bpm_ig_pos_law_stance
				subtract = scope:second_ig.bpm_ig_pos_law_stance
				divide = 2
				max = 1
				min = -1
				if = {
					limit = {
						scope:first_ig.bpm_ig_pos_law_stance < scope:second_ig.bpm_ig_pos_law_stance
					}
					multiply = -1
				}
				owner = {
					divide = {
						value = "variable_map(bpm_lawgroup_size|scope:law.type)"
						min = 1
					}
				}
				multiply = bpm_aggregate_salience
			}
		}
	}
}

bpm_inter_ig_approval_law_group = {
	value = 0
	save_temporary_scope_as = first_ig
	scope:ig = {
		save_temporary_scope_as = second_ig
	}
	owner = {
		every_law = {
			limit = {
				type = { save_temporary_scope_as = law_type }
				scope:law_type = {
					is_same_law_group_as = scope:law_group_ref.type
				}
				owner = {
					NOT = { has_law = prev.type }
				}
			}
			save_temporary_scope_as = law
			subtract = {
				value = scope:first_ig.bpm_ig_pos_law_stance
				subtract = scope:second_ig.bpm_ig_pos_law_stance
				divide = 2
				max = 1
				min = -1
				if = {
					limit = {
						scope:first_ig.bpm_ig_pos_law_stance < scope:second_ig.bpm_ig_pos_law_stance
					}
					multiply = -1
				}
				owner = {
					divide = {
						value = "variable_map(bpm_lawgroup_size|scope:law.type)"
						min = 1
					}
				}
				multiply = bpm_aggregate_salience
			}
		}
	}
}

bpm_ig_pos_law_stance = {
	value = 0
	if = {
		limit = {
			law_enactment_stance = {
				law = scope:law.type
				value = disapprove
			}
		}
		add = 1
	}
	else_if = {
		limit = {
			law_enactment_stance = {
				law = scope:law.type
				value = strongly_disapprove
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			law_enactment_stance = {
				law = scope:law.type
				value = approve
			}
		}
		add = 3
	}
	else_if = {
		limit = {
			law_enactment_stance = {
				law = scope:law.type
				value = strongly_approve
			}
		}
		add = 4
	}
}

bpm_consensus_on_lawgroup = {
	save_temporary_scope_as = lawgroup
	owner = {
		every_law = {
			limit = {
				scope:lawgroup.type = {
					is_same_law_group_as = prev.type
				}
			}
			save_temporary_scope_as = law
			add = owner.bpm_consensus_on_law
		}
	}
}

bpm_consensus_on_law = {
	value = bpm_consensus_on_law_raw
	if = {
		limit = {
			bpm_consensus_on_law_raw < 0
		}
		multiply = -1
	}
	min = 0
	max = 1
}

bpm_consensus_on_law_raw = {
	value = 0
	every_interest_group = {
		limit = {
			bpm_interest_group_is_activated = yes
		}
		if = {
			limit = {
				law_stance = {
					law = scope:law.type
					value > neutral
				}
			}
			add = ig_clout
		}
		else_if = {
			limit = {
				law_stance = {
					law = scope:law.type
					value < neutral
				}
			}
			subtract = ig_clout
		}
	}
	max = 1
	min = -1
}
