bpm_friendly_action_clout = {
	every_in_list = {
		variable = bpm_actively_involved
		if = {
			limit = {
				NOT = { scope:ig = this }
				bpm_ig_is_friendly = yes
			}
			add = ig_clout
		}
	}
	if = {
		limit = { bpm_total_action_clout > 0 }
		divide = bpm_total_action_clout
	}
}

# this = ig
bpm_ig_attraction_to_take_down_coalition = {
	value = 0
	save_temporary_scope_as = ig
	add = {
		value = 1
		desc = "bpm_attraction_take_down_base"
	}
	if = {
		limit = { NOT = { exists = owner.var:bpm_ruling_ig } }
		multiply = {
			value = 0
			desc = "bpm_attraction_reason_no_rig"
		}
	}
	else = {
		if = {
			limit = { owner.var:bpm_ruling_ig = this }
			multiply = {
				value = 0
				desc = "bpm_attraction_reason_is_rig"
			}
		}
		if = {
			limit = {
				ig_clout > owner.var:bpm_ruling_ig.ig_clout
				bpm_ig_attraction_to_take_over_rig > 0
			}
			multiply = {
				value = 0
				desc = "bpm_attraction_reason_more_clout_than_rig"
			}
		}
	}
	if = {
		limit = { owner = { has_variable = bpm_take_down_coalition_cooldown } }
		multiply = {
			value = 0
			desc = "bpm_attraction_reason_cooldown"
		}
	}
	if = {
		limit = { ig_approval > unhappy }
		multiply = {
			value = 0
			desc = "bpm_attraction_reason_not_unhappy"
		}
	}
	if = {
		limit = {
			is_powerful = yes
			bpm_ig_attraction_to_take_over_rig > 0
		}
		multiply = {
			value = 0
			desc = "bpm_attraction_reason_powerful"
		}
	}
}

				desc = "bpm_attraction_reason_cooldown"

			}

# this = ig
bpm_ig_attraction_to_negotiate_budget = {
	value = 0
	save_temporary_scope_as = ig
	if = {
		limit = {
			owner.var:bpm_ruling_ig ?= this
			ig_approval <= unhappy
		}
		add = {
			value = bpm_negotiate_budget_attraction_per_grievance
			desc = "bpm_attraction_negotiate_budget_unhappy_rig"
		}
	}
	if = {
		limit = { bpm_ig_consumption_tax_burden > 0 }
		add = {
			value = bpm_negotiate_budget_attraction_per_grievance
			desc = "bpm_attraction_negotiate_budget_consumption_taxes"
		}
	}
	if = {
		limit = { owner = { bpm_country_has_big_deficit = yes } }
		add = {
			value = bpm_negotiate_budget_attraction_per_grievance
			desc = "bpm_attraction_negotiate_budget_deficit"
		}
	}
	if = {
		limit = {
			bpm_ig_is_bourgeois = yes
			owner = { bpm_country_has_large_debt = yes }
		}
		add = {
			value = bpm_negotiate_budget_attraction_per_grievance
			desc = "bpm_attraction_negotiate_budget_debt"
		}
	}
	if = {
		limit = {
			NOT = { owner.var:bpm_ruling_ig ?= this }
			is_in_government = no
			ig_approval > angry
			ig_approval <= unhappy
			bpm_ig_attraction_to_take_down_coalition <= 0
			bpm_ig_attraction_to_take_over_rig <= 0
		}
		add = {
			value = bpm_negotiate_budget_attraction_unhappy_opposition
			desc = "bpm_attraction_negotiate_budget_unhappy_opposition"
		}
	}
	if = {
		limit = { owner = { has_variable = bpm_negotiate_budget_cooldown } }
		multiply = {
			value = 0
			desc = "bpm_attraction_reason_budget_cooldown"
		}
	}
}

bpm_ig_take_over_rig_support = {
	value = 0
	save_temporary_scope_as = bpm_support_ig
	if = {
		limit = { owner.var:bpm_ruling_ig ?= this }
		add = {
			value = -2
			desc = "bpm_ig_coalition_support_ruling_ig"
		}
	}
	else_if = {
		limit = { owner.var:bpm_proposed_ruling_ig ?= this }
		add = {
			value = 2
			desc = "bpm_ig_take_over_support_proposed_rig"
		}
	}
	else = {
		if = {
			limit = {
				owner = {
					has_variable_map = bpm_proposed_rig_approval_cache
					is_key_in_variable_map = {
						name = bpm_proposed_rig_approval_cache
						target = scope:bpm_support_ig
					}
				}
			}
			if = {
				limit = {
					owner = { "variable_map(bpm_proposed_rig_approval_cache|scope:bpm_support_ig)" >= 0.75 }
				}
				add = {
					value = 1
					desc = "bpm_ig_take_over_support_proposed_friendly"
				}
			}
			else_if = {
				limit = {
					owner = { "variable_map(bpm_proposed_rig_approval_cache|scope:bpm_support_ig)" < 0.5 }
				}
				add = {
					value = -1
					desc = "bpm_ig_take_over_support_proposed_hostile"
				}
			}
		}
		if = {
			limit = {
				owner = {
					has_variable_map = bpm_coalition_rig_approval_cache
					is_key_in_variable_map = {
						name = bpm_coalition_rig_approval_cache
						target = scope:bpm_support_ig
					}
				}
			}
			if = {
				limit = {
					owner = { "variable_map(bpm_coalition_rig_approval_cache|scope:bpm_support_ig)" >= 0.75 }
				}
				add = {
					value = -1
					desc = "bpm_ig_coalition_support_friendly"
				}
			}
			else_if = {
				limit = {
					owner = { "variable_map(bpm_coalition_rig_approval_cache|scope:bpm_support_ig)" < 0.5 }
				}
				add = {
					value = 1
					desc = "bpm_ig_coalition_support_hostile"
				}
			}
		}
		if = {
			limit = { ig_approval <= unhappy }
			add = {
				value = 1
				desc = "bpm_ig_take_down_support_unhappy"
			}
		}
		else_if = {
			limit = { ig_approval >= happy }
			add = {
				value = -1
				desc = "bpm_ig_take_over_support_happy"
			}
		}
	}
}
