bpm_setup_movement_panel = {
	scope = political_movement

	effect = {
		save_scope_as = movement
		clear_variable_list = bpm_ig_view
		owner = {
			every_supportive_interest_group = {
				prev = {
					add_to_variable_list = {
						name = bpm_ig_view
						target = prev
					}
				}
			}
		}
	}
}

bpm_cell_is_growing = {
	scope = state_region

	saved_scopes = {
		action
	}

	is_shown = {
		scope:action.parent = { save_temporary_scope_as = country }
		OR = {
			bpm_revcell_below_front_line = yes
			AND = {
				bpm_revcell_above_front_line = no
				bpm_cell_growth_chance > bpm_cell_shrinking_chance
			}
		}
	}
	
	effect = {
		scope:action.parent = { save_scope_as = country }
		bpm_increase_or_shrink_tt = yes
	}
}
bpm_revolt_select_action = {
	scope = country

	saved_scopes = {
		action
	}

	is_shown = {
		var:bpm_revolt_selected_action ?= scope:action
	}

	effect = {
		set_variable = {
			name = bpm_revolt_selected_action
			value = scope:action
		}
		bpm_update_grasp_lists = yes
	}
}
