ig_reactionaries = {
	color = hsv{ 0.67 0.7 0.9 }
	texture = "gfx/interface/icons/ig_icons/reactionaries.dds"
	layer = "revolution_dynamic_landowners"
	index = 14
	
	ideologies = {
		# Vanilla
		ideology_reactionary
		ideology_hierarchic
		ideology_patriotic
	}

	character_ideologies = {
		ideology_center_leader
	}
	
	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_name_reactionaries = yes
		bpm_update_ideology_reactionaries = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:USA ?= this
				any_primary_culture = { 
					cu:dixie = this
				}
			}
			cultures = {
				dixie
			}
		}

		rule = {
			trigger = {
				c:AUS ?= this
				any_primary_culture = { 
					cu:south_german = this
				}
			}
			cultures = {
				south_german
			}
		}

		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				british scottish irish
			}
		}
	}

	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				has_technology_researched = rationalism
				ig:ig_reactionaries ?= {
					bpm_interest_group_is_activated = yes
				}
			}
		}
	}
	
	pop_weight = {
		value = 1
	}
	
	monarch_weight = {
		owner.ig:ig_reactionaries = {
			value = bpm_ig_monarch_weight
		}
	}
	
	agitator_weight = {
		owner.ig:ig_reactionaries = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_reactionaries = {
			value = bpm_commander_weight
		}
	}
	
	executive_weight = {
		value = 0.1
		if ={
			limit = {
				owner.ig:ig_reactionaries = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}

	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
