bpm_rigidity_is_chaotic = {
	bpm_political_stability < bpm_rigidity_unstable_threshold
}

bpm_rigidity_is_unstable = {
	bpm_political_stability >= bpm_rigidity_unstable_threshold
	bpm_political_stability < bpm_rigidity_dynamic_threshold
}

bpm_rigidity_is_dynamic = {
	bpm_political_stability >= bpm_rigidity_dynamic_threshold
	bpm_political_stability <= bpm_rigidity_entrenched_threshold
}

bpm_rigidity_is_entrenched = {
	bpm_political_stability > bpm_rigidity_entrenched_threshold
	bpm_political_stability <= bpm_rigidity_stagnant_threshold
}

bpm_rigidity_is_stagnant = {
	bpm_political_stability > bpm_rigidity_stagnant_threshold
}

bpm_rigidity_at_most_unstable = {
	bpm_political_stability < bpm_rigidity_dynamic_threshold
}

bpm_rigidity_at_most_dynamic = {
	bpm_political_stability <= bpm_rigidity_entrenched_threshold
}

bpm_rigidity_at_least_entrenched = {
	bpm_political_stability > bpm_rigidity_entrenched_threshold
}
