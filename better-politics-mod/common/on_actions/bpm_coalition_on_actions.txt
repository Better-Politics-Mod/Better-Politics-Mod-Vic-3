on_monthly_pulse_country = {
	on_actions = {
		bpm_monthly_coalition_oa
		bpm_rig_approval_cache_monthly
		bpm_angry_coalition_members_monthly
	}
}

bpm_angry_coalition_members_monthly = {
	effect = {
		bpm_angry_coalition_members_countdown = yes
	}
}


on_game_started = {
	on_actions = {
		bpm_on_game_started_coalitions
	}
}

bpm_monthly_coalition_oa = {
	effect = {
		if = {
			limit = {
				is_player = no
			}
			bpm_coalition_and_actions = yes
		}
	}
}

bpm_on_game_started_coalitions = {
	effect = {
		bpm_coalition_and_actions_init = yes
	}
}

on_law_enactment_started = {
	on_actions = {
		bpm_rig_approval_cache_on_enactment_started
	}
}

on_law_enactment_ended = {
	on_actions = {
		bpm_rig_approval_cache_on_enactment_ended
	}
}

bpm_rig_approval_cache_on_enactment_started = {
	effect = {
		if = {
			limit = {
				currently_enacting_law.type ?= law_type:law_repeatable_special_take_over_rig
			}
			bpm_take_over_rig_enactment_started = yes
		}
		if = {
			limit = {
				currently_enacting_law.type ?= law_type:law_repeatable_special_take_down_coalition
			}
			set_variable = bpm_take_down_coalition_enacting
		}
		if = {
			limit = {
				currently_enacting_law.type ?= law_type:law_repeatable_special_negotiate_budget
			}
			set_variable = bpm_negotiate_budget_enacting
		}
		bpm_update_enactment_approval_caches = yes
	}
}

bpm_rig_approval_cache_on_enactment_ended = {
	effect = {
		bpm_clear_rig_approval_cache = yes
		bpm_clear_proposed_rig_approval_cache = yes
		if = {
			limit = { has_variable = bpm_take_over_rig_enacting }
			bpm_take_over_rig_enactment_ended = yes
		}
		if = {
			limit = { has_variable = bpm_take_down_coalition_enacting }
			bpm_take_down_coalition_enactment_ended = yes
		}
		if = {
			limit = { has_variable = bpm_negotiate_budget_enacting }
			bpm_negotiate_budget_enactment_ended = yes
		}
	}
}

bpm_rig_approval_cache_monthly = {
	effect = {
		bpm_update_enactment_approval_caches = yes
	}
}
