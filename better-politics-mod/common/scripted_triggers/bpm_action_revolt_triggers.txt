bpm_has_revcell = {
	scope:action = {
		is_key_in_variable_map = {
			name = bpm_revcells
			target = prev
		}
	}
}

bpm_has_revcell_level = {
	scope:action = {
		is_key_in_variable_map = {
			name = bpm_revcells
			target = prev
		}
		"variable_map(bpm_revcells|prev)" = $lvl$
	}
}

bpm_has_revolt_action = {
	any_in_list = {
		variable = bpm_active_actions
		var:bpm_action_type ?= flag:bpm_action_revolt
	}
}

bpm_revolt_map_cell_level = {
	state_region = {
		save_temporary_scope_as = sr
	}
	owner = {
		any_in_list = {
			variable = bpm_active_actions
			is_key_in_variable_map = {
				name = bpm_revcells
				target = scope:sr
			}
			"variable_map(bpm_revcells|scope:sr)" = $lvl$
		}
	}
}

bpm_revcell_can_be_spread_to = {
	any_scope_state = { turmoil > 0.25 }
	NOT = {
		bpm_has_revcell_level = { lvl = 4 }
	}
}

bpm_movement_is_broadly_modernizing = {
	OR = {
		law_stance = {
			law = law_type:law_separated_powers
			value > neutral
		}
		law_stance = {
			law = law_type:law_universal_suffrage
			value > neutral
		}
		law_stance = {
			law = law_type:law_census_voting
			value > neutral
		}
		law_stance = {
			law = law_type:law_presidential_republic
			value > neutral
		}
		AND = {
			owner = {
				has_law = law_type:law_serfdom
			}
			law_stance = {
				law = law_type:law_peasant_proprietorship
				value > neutral
			}
		}
	}
}

bpm_movement_is_broadly_proletarian = {
	OR = {
		law_stance = {
			law = law_type:law_factory_councils
			value > neutral
		}
		law_stance = {
			law = law_type:law_council_republic
			value > neutral
		}
	}
}

bpm_movement_is_broadly_reactionary = {
	NOR = {
		bpm_movement_is_broadly_modernizing = yes
		bpm_movement_is_broadly_proletarian = yes
	}
}

	save_temporary_scope_as = cell_sr
