enactment_can_sway = {
	scope = interest_group
	is_valid = {
		owner = {
			has_variable = bpm_enactment_debug_sway_selected
		}
	}
}

enactment_sway_ig = {
	scope = interest_group
	effect = {
		if = {
			limit = { NOT = { has_variable = bpm_support_from_sways } }
			set_variable = {
				name = bpm_support_from_sways
				value = 0
			}
		}
		if = {
			limit = { owner = { has_variable = bpm_debug_for } }
			change_variable = {
				name = bpm_support_from_sways
				add = 1
			}
		}
		else = {
			change_variable = {
				name = bpm_support_from_sways
				add = -1
			}
		}
		owner = { remove_variable = bpm_enactment_debug_sway_selected }
	}
}

enactment_debug_sway_for = {
	scope = country
	effect = {
		if = {
			limit = { has_variable = bpm_enactment_debug_sway_selected }
			remove_variable = bpm_enactment_debug_sway_selected
		}
		else = {
			set_variable = bpm_enactment_debug_sway_selected
		}
		set_variable = bpm_debug_for
	}
}

enactment_debug_sway_against = {
	scope = country
	effect = {
		if = {
			limit = { has_variable = bpm_enactment_debug_sway_selected }
			remove_variable = bpm_enactment_debug_sway_selected
		}
		else = {
			set_variable = bpm_enactment_debug_sway_selected
		}
		if = {
			limit = {
				has_variable = bpm_debug_for
			}
			remove_variable = bpm_debug_for
		}
	}
}
