REPLACE:reactionary_revolution_ongoing = {
	custom_tooltip = {
		text = reactionary_revolution_ongoing_tt
		any_political_movement = {
			is_revolutionary = yes
			OR = {
				any_influenced_interest_group = {
					ig_is_reactionary = yes
				}
				any_supporting_character = {
					has_role_of_type = agitator
					bpm_leader_is_reactionary = yes
				}
			}		
		}
	}	
}

REPLACE:ai_has_enact_weight_modifier_journal_entries = {
	OR = {
		has_law_commitment = scope:law.type
		AND = {
			has_journal_entry = je_government_petition
			has_variable = desired_law_var
			scope:law = var:desired_law_var
		}
		AND = {
			OR = {
				has_journal_entry = je_law_imposition
				has_journal_entry = je_british_dictate_law
			}
			scope:law = imposed_law
		}
		AND = {
			has_journal_entry = je_gyojo_shinwon
			OR = {
				AND = {
					exists = scope:donghak_desired_law_1
					scope:law = scope:donghak_desired_law_1
				}
				AND = {
					exists = scope:donghak_desired_law_2
					scope:law = scope:donghak_desired_law_2
				}
				AND = {
					exists = scope:donghak_desired_law_3
					scope:law = scope:donghak_desired_law_3
				}
			}
		}
	}
}
