bpm_ig_relation_color_sg = {
	scope = interest_group
	saved_scopes = {
		ig
	}

	effect = {
		if = {
			limit = {
				bpm_inter_ig_approval < 0.5
			}
			custom_tooltip_no_bullet = bpm_relation_color_red
		}
		else_if = {
			limit = {
				bpm_inter_ig_approval < 0.75
			}
			custom_tooltip_no_bullet = bpm_relation_color_grey
		}
		else = {
			custom_tooltip_no_bullet = bpm_relation_color_green
		}
	}
}
