ig_conservatives = {
	color = hsv{ 0.57 1 1 }
	texture = "gfx/interface/icons/ig_icons/conservatives.dds"
	layer = "revolution_dynamic_petty_bourgeoisie"
	index = 10

	ideologies = {
		# Vanilla
		ideology_paternalistic
		ideology_hierarchic
		ideology_patriotic
	}
	
	character_ideologies = {
		ideology_center_leader
		ideology_traditionalist
		ideology_jingoist_leader
		ideology_protectionist
		ideology_sovereignist_leader
		ideology_corporatist_leader
	}

	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_conservatives = yes
		bpm_update_name_conservatives = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				assamese avadhi baluchi bengali bihari gujarati kannada malayalam marathi oriya panjabi pathan rajput sindi tamil telegu hindustani bundeli pahari gondi bageli chhattisgarhi deccani
			}
		}
	}

	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				ig:ig_conservatives ?= {
					bpm_interest_group_is_activated = yes
				}
			}
		}
	}
	
	pop_weight = {
		value = 0
		
		# Main intended support
		
		if = {
			limit = {
				is_pop_type = capitalists
			}
			add = {
				desc = "POP_CAPITALISTS"
				value = 20
			}
			add = {
				desc = "LIBERAL_CONSCIOUSNESS"
				value = owner.bpm_consciousness_liberal_igs_capitalists_shopkeepers
			}
		}

		# General support

		

		if = {
			limit = { 
				strata = lower 
			}
			add = {
				desc = "LOWER_CLASS"
				value = 20
			}
		}
		else_if = {
			limit = { 
				strata = middle 
			}
			add = {
				desc = "MIDDLE_CLASS"
				value = 50
			}
		}
		else_if = {
			limit = {
				is_pop_type = capitalists
			}
			add = {
				desc = "POP_CAPITALISTS"
				value = 20
			}
		}
		else_if = {
			limit = {
				is_pop_type = aristocrats
			}
			add = {
				desc = "POP_ARISTOCRATS"
				value = 50
			}
			add = state.modifier:state_bpm_ig_conservatives_aristocrats_attraction_weight_add
		}

		# RURAL
		if = {
			limit = {
				bpm_pop_employment_is_rural = yes
			}
			add = {
				desc = "RURAL"
				value = 50
			}
		}

		# PATCON AND URBAN
		else_if = {
			limit = {
				scope:interest_group = { 
					bpm_ig_is_paternalistic = yes
				}
			}
			if = {
				limit = { strata = lower }
				add = {
					desc = "PATCON_LOWER_CLASS"
					value = 100
				}
			}
			else_if = {
				limit = { strata = middle }
				add = {
					desc = "PATCON_MIDDLE_CLASS"
					value = 20
				}
			}
			else_if = {
				limit = { strata = upper }
				add = {
					desc = "PATCON_CAPITALISTS"
					value = 10
				}
			}
		}

		add = {
			desc = "EDUCATED"
			value = literacy_rate
			multiply = 150
			max = 50
		}

		owner = {
			multiply = {
				desc = "SPLIT_MORALISTS"
				value = 1.0

				if = {
					limit = {
						has_variable = local_christdem_split_var
						has_global_variable = global_rerum_novarum_issued_var
						owner = {
							ig:ig_conservatives = {
								ig_clout <= 0.05
							}
						}
					}
					value = 0.5
				}			
			}
		}

		multiply = {
			desc = "STILL_OLIGARCHIC"
			value = 1.0
			if = {
				limit = {
					owner = {
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_monarchy
					}
				}
				value = 0.5
			}
		}
		multiply = ig_attraction_from_leader_popularity
		multiply = bpm_ig_agitators_popularity
		multiply = bpm_ig_ruler_popularity
		multiply = { desc = bpm_ig_attraction_from_radicals_and_loyalists_tt value = ig_attraction_from_radicals_and_loyalists }
		multiply = ig_attraction_to_ruler_ideology_from_stability
		multiply = ig_attraction_from_government

		multiply = {
			desc = "AT_WAR"	
			value = 1.0

			if = {
				limit = { owner = { is_at_war = yes } }
				value = 1.4
			}		
		}
	}
	
	monarch_weight = {
		owner.ig:ig_conservatives = {
			value = bpm_ig_monarch_weight
		}
	}
	
	agitator_weight = {
		owner.ig:ig_conservatives = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_conservatives = {
			value = bpm_commander_weight
		}
	}

	executive_weight = {
		value = 0.3
		if ={
			limit = {
				owner.ig:ig_conservatives = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
