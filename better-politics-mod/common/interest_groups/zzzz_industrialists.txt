REPLACE:ig_industrialists = {
	color = hsv{ 0.08 0.2 0.55 }
	texture = "gfx/interface/icons/ig_icons/industrialists.dds"
	layer = "revolution_dynamic_industrialists"
	index = 2

	ideologies = {
		# Vanilla
		ideology_plutocratic
		ideology_laissez_faire
		ideology_individualist
	}

	character_ideologies = {
		ideology_moderate
		ideology_protectionist
	}
	
	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_industrialists = yes
		bpm_update_name_industrialists = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				british scottish irish
			}
		}
		rule = {
			trigger = {
				c:ZAN ?= this
				any_primary_culture = {
					cu:swahili = this
				}
			}
			cultures = {
				bedouin swahili north_german british gujarati
			}
		}
		rule = {
			trigger = {
				c:HAW ?= this
				NOT = { has_variable = deported_americans }
			}
			cultures = {
				yankee dixie british hawaiian
			}
		}
	}


	on_character_ig_membership = {
		if = {
			limit = {
				OR = {
					culture = cu:british
					culture = cu:north_german
				}
				owner = {
					c:ZAN ?= this
					any_primary_culture = {
						cu:swahili = this
					}
				}
			}
			change_character_religion = rel:protestant
		}
		if = {
			limit = {
				culture = cu:gujarati
				owner = {
					c:ZAN ?= this
					any_primary_culture = {
						cu:swahili = this
					}
				}
			}
			change_character_religion = rel:shiite
		}
	}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				NOT = {
					is_country_type = decentralized
				}
			}
		}
		OR = {
			is_pop_type = capitalists
			is_pop_type = engineers
			is_pop_type = shopkeepers
			is_pop_type = aristocrats
			#is_pop_type = bureaucracts
			AND = {
				exists = c:BIC
				owner ?= c:BIC
				OR = {
					culture ?= {
						has_discrimination_trait_group = heritage_group_european
					}
					is_pop_type = bureaucrats
					is_pop_type = clerks
				}
			}
		}
	}
	
	pop_weight = {
		value = 0
	
		if = {
			limit = { is_pop_type = capitalists }
			add = {
				desc = "POP_CAPITALISTS"
				value = 100
			}
			add = state.modifer:state_bpm_ig_industrialists_capitalists_attraction_weight_add
			if = {
				limit = {
					owner = { 
						has_variable = bpm_liberal_movement 
						NOT = { exists = ig:ig_liberals }
					}
				}
				add = {
					desc = "POP_CAPITALISTS"
					value = 50
				}
			}
		}
		else_if = {
			limit = { 
				is_pop_type = aristocrats
				pop_employment_building = building_company_headquarter
			}
			add = {
				desc = "POP_ARISTOCRATS"
				value = 100
			}
		}
		else_if = {
			limit = { is_pop_type = engineers }
			if = {
				limit = {
					always = yes # the CR support thing here
				}
			}
			else = {
				add = {
					desc = "POP_ENGINEERS"
					value = 10
				}
			}
		}
		else_if = {
			limit = { is_pop_type = shopkeepers }
			add = {
				desc = "POP_SHOPKEEPERS"
				value = 20
			}
		}
		
		if = {
			limit = {
				owner = {
					c:BIC ?= this
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
				OR = {
					strata = upper
					strata = middle
				}
				NOT = {
					is_pop_type = officers
				}
			}
			add = {
				desc = "POP_INDIA_EUROPEAN_ADMINISTRATORS"
				value = 1000 #250
			}
		}
		
		if = {
			limit = {
				owner = {
					c:BIC ?= this
				}
			}
			if = {
				limit = {
					NOT = {
						culture = {
							has_discrimination_trait_group = heritage_group_european
						}
					}
					OR = {
						is_pop_type = bureaucrats
						is_pop_type = clerks
					}
				}
				add = {
					desc = "POP_INDIA_NATIVE_ADMINISTRATORS"
					value = 100 #25
				}
			}
			else_if = {
				limit = {
					NOT = {
						culture = {
							has_discrimination_trait_group = heritage_group_european
						}
					}
					NOR = {
						is_pop_type = bureaucrats
						is_pop_type = clerks
					}
				}
				multiply = {
					desc = "POP_INDIA_ADMINISTRATION_PRIVILEGES"
					value = 0.5 #0.05
				}
			}
		}

		add = {
			desc = "NOT_RURAL_POP"
			if = {
				limit = { 
					bpm_pop_employment_is_rural = yes
				}
				value = -50
			}
		}

		multiply = ig_attraction_to_interests_from_stability
		multiply = ig_attraction_from_government
		multiply = ig_attraction_from_leader_popularity
		
	}
	
	monarch_weight = {
		owner.ig:ig_industrialists = {
			value = bpm_ig_monarch_weight
		}
	}	

	agitator_weight = {
		owner.ig:ig_industrialists = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_industrialists = {
			value = bpm_commander_weight
		}
	}	

	executive_weight = { # MAYBE make some sort of bureaucratic - industrialist mix under command eco, with bureacurats supporting indust and the reverse
		value = 2.0
		if = {
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_cooperative_ownership
						has_law_or_variant = law_type:law_command_economy
					}
				}
			}
			value = 0.0
		}
	}
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
