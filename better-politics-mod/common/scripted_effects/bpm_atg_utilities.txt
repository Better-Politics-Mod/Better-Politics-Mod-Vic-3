bpm_update_government_and_opposition = {
	bpm_add_ig_to_their_desired_party = yes
	if = {
		limit = {
			modifier:country_government_formation_type_popular = yes
		}
		bpm_form_government_all_systems = yes
	}
	else_if = {
		limit = {
			modifier:country_government_formation_type_traditional = yes
		}
		bpm_form_government_all_systems = yes
	}
	else_if = {
		limit = {
			modifier:country_government_formation_type_institutional = yes
		}
		bpm_form_government_all_systems = yes
	}
}

bpm_form_government_all_systems = {
	every_interest_group = {
		add_ruling_interest_group = yes
	}
	every_interest_group = {
		limit = {
			NOR = {
				AND = {
					exists = party
					owner.var:bpm_headofgov.party ?= party
				}
				owner.var:bpm_headofgov ?= this
			}
		}
		remove_ruling_interest_group = yes
	}
}

bpm_add_ig_to_their_desired_party = {
	update_party_support = yes
	every_interest_group = {
		limit = {
			NOT = {
				exists = party
			}
			is_marginal = no
		}
		add_ruling_interest_group = yes
	}
	update_party_support = yes
	every_interest_group = {
		limit = {
			NOT = {
				exists = party
			}
			is_marginal = no
		}
		remove_ruling_interest_group = yes
	}
	update_party_support = yes
}

bpm_get_picked_pm_util = {
	save_scope_value_as = { name = counter value = 0 }

		variable = bpm_potential_headofgov
		if = { 
			limit = { scope:counter = owner.var:bpm_choose_headofgov_carousel }
			save_scope_as = result 
		}
		save_scope_value_as = {
			name = counter
			value = {
				value = 1
				add = scope:counter
			}
		}
	}
}
