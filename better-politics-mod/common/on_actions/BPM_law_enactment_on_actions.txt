on_law_enactment_ended = {
	on_actions = {
		on_bpm_law_enactment_ended
	}
}

on_law_activated = {
	on_actions = {
		on_bpm_law_enactment_completed
	}
}

on_law_enactment_fail = {
	on_actions = {
		on_bpm_law_enacted_fail
	}
}

on_bpm_law_enactment_ended = {
	effect = {
		bpm_the_docket_cooldown = yes
	}
}

on_bpm_law_enactment_completed = {
	effect = {		
		owner = { bpm_update_law_enactment_docket = yes }
	}
}

on_bpm_law_enacted_fail = {
	effect = {
		trigger_event = { id = bpm_misc.99 days = 1 }
	}
}

# Root = owner Country of the Law
# Executed when the end of a checkpoint is reached and the result is success
on_law_checkpoint_success = {
	effect = {
		if = {
			limit = {
				enacting_any_law = yes
			}
			currently_enacting_law = {
				save_scope_as = law
			}
			if = {
				limit = { is_player = yes }
				if = {
					limit = { enactment_phase = 2 }
					if = {
						limit = { bpm_country_yea_votes >= bpm_law_enactment_vote_threshold }
						if = {
							limit = { bpm_law_enactment_executive_must_accept = yes }
							save_scope_as = associated_country
							bpm_snapshot_take_over_rig_supporters = yes
							if = {
								limit = {
									currently_enacting_law.type ?= law_type:law_repeatable_special_form_coalition
								}
								bpm_form_ruling_coalition = yes
							}
							add_enactment_phase = 1
							
						}
						else = {
							trigger_event = { id = bpm_law_veto.1 popup = yes }
						}
					}
					else = {
						add_enactment_setback = 1
						if = {
							limit = {
								OR = {
									enacting_any_law = no
									enactment_setback_count = 3
								}
							}
							trigger_event = { id = unable_to_enact.2 popup = yes }
						}
					}
				}
				else_if = {
					limit = { enactment_phase = 1 }
					add_enactment_phase = 1
					trigger_event = { id = law_notification.4 days = 0 }
				}
				else = {
					add_enactment_phase = 1
					trigger_event = { id = law_notification.4 days = 0 }
				}
			}
			else = {
				bpm_snapshot_take_over_rig_supporters = yes
				if = {
					limit = {
						currently_enacting_law.type ?= law_type:law_repeatable_special_form_coalition
					}
					bpm_form_ruling_coalition = yes
				}
				add_enactment_phase = 1
				if = {
					limit = { enacting_any_law = yes }
					trigger_event = { id = law_notification.4 days = 0 }
				}
			}
		}
	}
}
