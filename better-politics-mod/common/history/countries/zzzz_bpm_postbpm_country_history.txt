COUNTRIES = {
    every_country = {
        if = {
            limit = {
                AND = {
                    is_in_geographic_region = geographic_region_new_world
                    bpm_country_is_bourgeois_republic = yes
                }
            }
            active_law:lawgroup_governance_principles = {
                add_amendment = {
                    type = bpm_amendment_republican_tradition
                    sponsor = prev.ig:ig_petty_bourgeoisie
                }
            }
        }

        if = {
			limit = { 
				bpm_country_is_pre_constitutional = yes
				OR = {
					has_law_or_variant = law_type:law_freedom_of_conscience
					has_law_or_variant = law_type:law_state_religion
				}
			}
			active_law:lawgroup_education_system ?= { # No Freedom of Education
				add_amendment = {
					type = bpm_amendment_no_freedom_of_education
					sponsor = PREV.ig:ig_devout
				}
			}
		}

		if = {
			limit = { 
				OR = {
					has_law = law_type:law_no_police
					has_law = law_type:law_local_police
				}
				bpm_country_is_pre_constitutional = yes
				NOT = {is_in_geographic_region = geographic_region_new_world}
			}
			active_law:lawgroup_policing ?= {
				add_amendment = {
					type = bpm_amendment_noble_police_privilages
					sponsor = PREV.ig:ig_landowners
				}
			}
		}

		if = {
			limit = {
				bpm_country_is_pre_constitutional = yes
			}
			active_law:lawgroup_internal_security ?= {
				add_amendment = {
					type = bpm_amendment_inequality_before_the_law
					sponsor = PREV.ig:ig_landowners
				}
			}
		}
		
		if = {
			limit = { 
				OR = {
					has_law = law_type:law_land_based_taxation
					has_law = law_type:law_tenant_farmers
				}
				has_law = law_type:law_no_labour_law
				NOT = {is_in_geographic_region = geographic_region_new_world}
			}
			active_law:lawgroup_taxation ?= {
				add_amendment = {
					type = bpm_amendment_peasant_taxes
					sponsor = PREV.ig:ig_landowners
				}
			}
		}
		
		if = {
			limit = { 
				OR = {
					has_law_or_variant = law_type:law_tenant_farmers
					has_law_or_variant = law_type:law_serfdom
				}
				has_law = law_type:law_no_labour_law
				NOT = {is_in_geographic_region = geographic_region_new_world}
			}
            active_law:lawgroup_land_reform ?= {
				add_amendment = {
					type = bpm_amendment_noble_forests
					sponsor = PREV.ig:ig_landowners
				}
			}
		}
        
        if = {
            limit = {
                country_has_state_religion = rel:confucian
                has_law = law_type:law_monarchy
            }
            add_journal_entry = { type = je_bpm_confucian_state }
        }

        if = {
	    	limit = {
	    		has_law_or_variant = law_type:law_oligarchy
	    	}
	    	activate_law = law_type:law_autocracy
	    }
        bpm_setup_codification = yes

        bpm_activate_interest_group_initialize = yes
    }
}
