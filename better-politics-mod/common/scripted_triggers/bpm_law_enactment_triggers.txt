bpm_law_enactment_executive_must_accept = {
	enacting_any_law = yes
	OR = {
		trigger_if = {
			limit = {
				modifier:country_bpm_ruler_complete_veto_enabled = yes
			}
			always = no
		}
		trigger_else_if = {
			limit = {
				modifier:country_bpm_ruler_partial_veto_enabled = yes
			}
			bpm_country_yea_votes > 0.6666
		}
		trigger_else = {
			always = yes
		}
		has_modifier = bpm_enacting_no_confidence
		currently_enacting_law.type = law_type:law_repeatable_special_form_coalition
		currently_enacting_law.type = law_type:law_repeatable_special_take_down_coalition
		currently_enacting_law.type = law_type:law_repeatable_special_take_over_rig
	}
	NAND = {
		bpm_country_yea_votes < 0.66
		currently_enacting_law.currently_active_law_in_group ?= { bpm_law_is_constitutional = yes }
	}
}

strongly_supports_law = { #IG scope, checks currently enacting law
	law_enactment_stance = {
        law = owner.currently_enacting_law.type
        value > approve
    }
}

strongly_opposes_law = { #IG scope, checks currently enacting law
    law_enactment_stance = {
        law = owner.currently_enacting_law.type
        value < disapprove
    }
}

bpm_law_can_only_be_voted_on_by_state_institutions = {
	bpm_law_is_edict = yes
}

bpm_ig_can_vote_on_law = {
	trigger_if = {
		limit = { NOT = { exists = scope:law } }
		owner.currently_enacting_law = { save_temporary_scope_as = law }
	}
	
	trigger_if = {
		limit = {
			scope:law = {
				bpm_law_can_only_be_voted_on_by_state_institutions = yes
			}
		}
		bpm_ig_is_state_institutional = yes
	}
	trigger_else = {
		always = yes
	}
}

bpm_law_executive_would_edict = {
	save_temporary_scope_as = law
	owner = {
		modifier:country_bpm_executive_can_edict = yes
		bpm_executive_edict_support >= 10
	}
}

bpm_law_edict_would_be_illegal = {
	owner.modifier:country_bpm_law_edicts_illegal = yes
	NOR = {
		bpm_law_is_edict = yes
		bpm_law_is_constitutional = yes
	}
}

bpm_law_is_docket_candidate = {
	owner = {
		NOR = {
			has_law = prev.type
			currently_enacting_law.type ?= prev.type
			is_target_in_variable_list = { name = bpm_cooldown_laws target = prev.type }
		}
	}
	law_is_available = yes
	bpm_law_docket_score > 0
}
