bpm_set_ig_repealing_amendment_vars = {
	bpm_set_ig_parent_law_stance_to_number = yes
	bpm_set_ig_attached_law_stance_to_number = yes
	bpm_set_ig_repealing_amendment_change_steps = yes
}

bpm_set_ig_parent_law_stance_to_number = {
	set_variable = { name = bpm_ig_parent_law_stance value = 0 }
	
	if = {
		limit = { owner = { has_variable = bpm_repealing_amendment }}
		if = {
			limit = {
				law_stance = {
					law = owner.var:bpm_repealing_amendment.parent_law_type
					value = strongly_approve
				}
			}
			set_variable = { name = bpm_ig_parent_law_stance value = 2 }
		}
		else_if = {
			limit = {
				law_stance = {
					law = owner.var:bpm_repealing_amendment.parent_law_type
					value = approve
				}
			}
			set_variable = { name = bpm_ig_parent_law_stance value = 1 }
		}
		else_if = {
			limit = {
				law_stance = {
					law = owner.var:bpm_repealing_amendment.parent_law_type
					value = disapprove
				}
			}
			set_variable = { name = bpm_ig_parent_law_stance value = -1 }
		}
		else_if = {
			limit = {
				law_stance = {
					law = owner.var:bpm_repealing_amendment.parent_law_type
					value = strongly_disapprove
				}
			}
			set_variable = { name = bpm_ig_parent_law_stance value = -2 }
		}
	}
}

bpm_set_ig_repealing_amendment_change_steps = {
	set_variable = { 
		name = bpm_ig_repealing_amendment_change_steps
		value = {
			add = var:bpm_ig_attached_law_stance
			subtract = var:bpm_ig_parent_law_stance
		}
	}
