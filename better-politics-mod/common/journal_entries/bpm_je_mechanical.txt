
je_bpm_weekly_pulse = {
	dlc = dlc_bpm

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_bpm_misc_mechanical

	is_shown_when_inactive = {
		always = yes
	}

	possible = {
		always = yes
		# starting triggers
	}

	immediate = {
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			hidden_trigger = {
				always = no
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			owner = { 
				bpm_actions_weekly = yes 
				if = {
					limit = {
						is_player = yes
					}
					bpm_coalition_and_actions = yes
				}
			}
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = no
}


je_bpm_law_vote = {
	dlc = dlc_bpm

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_bpm_misc_mechanical

	is_shown_when_inactive = {
		always = yes
	}

	scaling_modifiers_while_active = {
		modifier = bpm_law_enactment_success_vote_contribution
		value = {
			if = {
				limit = { enacting_any_law = yes }
				value = bpm_law_enactment_success_vote_contribution
			}
			else = {
				value = 0

				always = no
			}

	should_be_pinned_by_default_uninvolved_or_context = no
}
