bpm_populate_lawparent_varmap = {
    random_decentralized_country = { # hopefully no one notices
        every_active_law = {
            owner = {
				add_to_variable_list = {
            	    name = bpm_original_laws
            	    target = prev
            	}
			}
        }
        every_law = {
            save_scope_as = law
            owner = {
                activate_law = prev.type
                random_law = {
                    limit = {
                        type = {save_temporary_scope_as = type}
                        owner = {has_law_or_variant = prev.type}
                        scope:type = {is_same_law_group_as = scope:law.type}
                        NOT = {
                            this = scope:law
                        }
                    }
                    alternative_limit = {
                        this = scope:law
                    }
                    add_to_global_variable_map = {
                        name = bpm_lawparent
                        key = scope:law.type
                        value = type
                    }
                }
            }
        }
        every_in_list = {
            variable = bpm_original_laws
            owner = {
                activate_law = prev.type
            }
        }
        clear_variable_list = bpm_original_laws
    }
}
