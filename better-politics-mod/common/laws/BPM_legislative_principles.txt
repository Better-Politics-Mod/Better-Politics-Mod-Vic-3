	progressiveness = 50
	
	modifier = {

		country_bureaucrats_pol_str_mult = 0.25
		country_bpm_auth_state_capacity_loss_from_governing_ig = 0.02
		country_bpm_auth_state_capacity_loss_from_bureaucratic_ig = 5
	}

	on_enact = {
		if = {
			limit = {
				has_journal_entry = je_springtime_of_the_peoples
			}
			bpm_springtime_all_three_revdrift_increase = yes
		}
	}

	can_enact = {
		custom_tooltip = {
			text = "bpm_law_enactment_requires_voting_system"
			country_has_voting_franchise = yes 
		}
	}

	disallowing_laws = {
		law_organic_democracy
		law_colonial_administration
		law_anarchy
	}

	on_activate = {
		if = {
			limit = {
				has_journal_entry = je_springtime_of_the_peoples
			}
			bpm_springtime_all_three_revdrift_increase = yes
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
		if = {
			limit = {
				has_law = law_type:law_advisory_body
				bpm_consciousness_liberal_percent > 0.35
			}
			add = 75
		}
	}
}
