
bpm_repeal_amendment_confirmation_gui = {
    scope = amendment

	confirm_title = bpm_repeal_amendment_title
	confirm_text = bpm_repeal_amendment_text
	
	is_valid = {
		custom_tooltip = { 
			text = "BPM_AMENDMENT_CAN_BE_REPEALED"
			bpm_amendment_can_be_repealed = yes
		}
		attached_law.owner = {
			enacting_any_law = no
		}
	}
    effect = {
		save_scope_as = amendment
		attached_law.owner = {
			set_variable = {
				name = bpm_repealing_amendment
				value = scope:amendment
			}
			set_variable = {
				name = bpm_repealing_amendment_law
				value = scope:amendment.attached_law
			}
			show_as_tooltip = {
				start_enactment = law_type:law_repeatable_politics_repeal_amendment_other
			}
			hidden_effect = {
				start_enactment = law_type:law_repeatable_politics_repeal_amendment
				add_enactment_phase = 2
			}
		}
    }
}

bpm_amendment_is_sticky_gui = {
	scope = amendment

	is_valid = {
		attached_law.owner = {
			any_in_list = {
				variable = bpm_sticky_amendments
				var:amendment_type = root.type
				var:sponsor = root.sponsor
				var:original_parent_type = { is_same_law_group_as = root.attached_law.type }
			}
		}
	}
}
