namespace = bpm_revsoc_utils

bpm_revsoc_utils.1 = {
	type = country_event
	placement = root
	hidden = yes

	trigger = {
		owner = {
			NOT = { has_law = law_type:law_council_republic }
			any_interest_group = {
				bpm_ig_is_revolutionary_socialist = yes
				NOT = { has_modifier = bpm_very_stubborn_ig }
			}
		}
	}

	immediate = {
		owner = {
			every_interest_group = {
				limit = {
					bpm_ig_is_revolutionary_socialist = yes
					NOT = { has_modifier = bpm_very_stubborn_ig }
				}
				add_modifier = { name = bpm_very_stubborn_ig }
			}
		}
	}
}
