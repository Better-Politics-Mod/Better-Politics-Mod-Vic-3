bpm_action_revolt_do = {
	scope:action = {
		ordered_in_list = {
			variable = bpm_actively_involved
			order_by = ig_clout
			save_scope_as = revolt_leader
		}
	}
	scope:country = {
		bpm_shrink_or_grow_in_states = yes
		bpm_revolt_post_toast = { list = bpm_revolt_toast_grows msg = bpm_revolution_grows }
		bpm_revolt_post_toast = { list = bpm_revolt_toast_spillover msg = bpm_revolution_spillover }
		bpm_revolt_post_toast = { list = bpm_revolt_toast_shrinks msg = bpm_revolution_shrinks }
		bpm_revolt_post_toast = { list = bpm_revolt_toast_subsides msg = bpm_revolution_subsides }
	}
	clear_saved_scope = revolt_leader
}

# this = state_region
# scope:country
bpm_revolt_queue_toast = {
	scope:country = {
		add_to_variable_list = {
			name = $list$
			target = prev
		}
	}
}

# this = country
bpm_revolt_post_toast = {
	if = {
		limit = { has_variable_list = $list$ }
		set_local_variable = {
			name = bpm_revolt_toast_n
			value = 0
		}
		every_in_list = {
			variable = $list$
			change_local_variable = {
				name = bpm_revolt_toast_n
				add = 1
			}
			if = {
				limit = { local_var:bpm_revolt_toast_n = 1 }
				save_scope_as = revolt_toast_1
			}
			else_if = {
				limit = { local_var:bpm_revolt_toast_n = 2 }
				save_scope_as = revolt_toast_2
			}
			else_if = {
				limit = { local_var:bpm_revolt_toast_n = 3 }
				save_scope_as = revolt_toast_3
			}
		}
		save_scope_value_as = {
			name = revolt_toast_count
			value = local_var:bpm_revolt_toast_n
		}
		save_scope_value_as = {
			name = revolt_toast_others
			value = {
				value = local_var:bpm_revolt_toast_n
				add = -3
			}
		}
		scope:revolt_toast_1 = {
			post_notification = $msg$
		}
		clear_variable_list = $list$
		remove_local_variable = bpm_revolt_toast_n
		clear_saved_scope = revolt_toast_1
		clear_saved_scope = revolt_toast_2
		clear_saved_scope = revolt_toast_3
		clear_saved_scope = revolt_toast_count
		clear_saved_scope = revolt_toast_others
	}
}

# revcell varmap -> determines the cell in the state
# revcell_progress varmap -> progress to up or down

bpm_shrink_or_grow_in_states = {
	every_key_in_variable_map = {
		variable = bpm_revcells
		if = {
			limit = {
				NOT = {
					exists = this
				}
			}
			owner = {
				remove_from_variable_map = {
					name = bpm_revcells
					key = prev
				}
			}
		}
	}
	if = {
		limit = {
			scope:action = {
				has_variable_map = bpm_revcells
			}
		}
		scope:action = {
			every_key_in_variable_map = {
				variable = bpm_revcells
				bpm_increase_or_shrink_cell_level = yes
			}
		}
		bpm_revolt_spawn_cells_below_front_line = yes
	}
	else = {
		random_scope_state = {
			limit = {
				turmoil >= 0.25
			}
			state_region = {
				bpm_increase_cell_level = yes
			}
		}
	}
}

# this = country
# scope:action
bpm_revolt_spawn_cells_below_front_line = {
	if = {
		limit = {
			scope:action = { has_variable = bpm_grasp_threshold }
		}
		every_scope_state = {
			limit = {
				turmoil >= 0.25
				bpm_action_cell_level = 0
				bpm_state_grasp_category < scope:action.var:bpm_grasp_threshold
			}
			random = {
				chance = bpm_revcell_front_line_chance
				state_region = {
					bpm_increase_revcell = yes
				}
			}
		}
	}
}

# Expects that current scope is a state region
bpm_increase_revcell = {
	if = {
		limit = {
			bpm_has_revcell = no
		}
		scope:action = {
			update_variable_map_value = {
				name = bpm_revcells
				key = prev
				value = 1
			}
		}
	}
	else = {
		scope:action = {
			set_local_variable = {
				name = new_level
				value = {
					value = "variable_map(bpm_revcells|prev)"
					add = 1
				}
			}
			update_variable_map_value = {
				name = bpm_revcells
				key = prev
				value = local_var:new_level
			}
			remove_local_variable = new_level
		}
	}
	if = {
		limit = { has_local_variable = bpm_revolt_is_spillover }
		bpm_revolt_queue_toast = { list = bpm_revolt_toast_spillover }
	}
	else_if = {
		limit = {
			NOT = { bpm_has_revcell_level = { lvl = 1 } }
		}
		bpm_revolt_queue_toast = { list = bpm_revolt_toast_grows }
	}
}

bpm_reduce_revcell = {
	if = {
		limit = {
			bpm_has_revcell_level = { lvl = 1 }
		}
		scope:action = {
			remove_from_variable_map = {
				name = bpm_revcells
				key = prev
			}
		}
		bpm_revolt_queue_toast = { list = bpm_revolt_toast_subsides }
	}
	else_if = {
		limit = {
			bpm_has_revcell = yes
		}
		bpm_revolt_queue_toast = { list = bpm_revolt_toast_shrinks }
		scope:action = {
			set_local_variable = {
				name = new_level
				value = {
					value = "variable_map(bpm_revcells|prev)"
					add = -1
				}
			}
			update_variable_map_value = {
				name = bpm_revcells
				key = prev
				value = local_var:new_level
			}
			remove_local_variable = new_level
		}
	}
}

bpm_reduce_cell_level = {
	bpm_reduce_revcell = yes
}

bpm_increase_cell_level_spillover = {
	set_local_variable = bpm_revolt_is_spillover
	bpm_increase_revcell = yes
	remove_local_variable = bpm_revolt_is_spillover
}

bpm_increase_cell_level = {
	if = {
		limit = {
			NOT = { bpm_has_revcell_level = { lvl = 4 } }
		}
		bpm_increase_revcell = yes
	}
	else_if = {
		limit = {
			bpm_has_revcell_level = { lvl = 4 }
			any_scope_state = {
				any_neighbouring_state = {
					state_region = {
						bpm_revcell_can_be_spread_to = yes
					}
				}
			}
		}
		random_scope_state = {
			limit = {
				any_neighbouring_state = {
					state_region = {
						bpm_revcell_can_be_spread_to = yes
					}
				}
			}
			random_neighbouring_state = {
				limit = {
					state_region = {
						bpm_revcell_can_be_spread_to = yes
					}
				}
				state_region = {
					bpm_increase_cell_level_spillover = yes
				}
			}
		}
	}
}

bpm_increase_or_shrink_tt = {
	if = {
		limit = { bpm_revcell_below_front_line = yes }
		custom_tooltip = "bpm_cell_grows_below_front_line"
	}
	else_if = {
		limit = { bpm_revcell_above_front_line = yes }
		custom_tooltip = "bpm_cell_shrinks_above_front_line"
	}
	else = {
		random_list = {
			0 = {
				modifier = {
					add = bpm_cell_shrinking_chance
				}
				custom_tooltip = "bpm_cell_shrinks"
			}
			0 = {
				modifier = {
					add = bpm_cell_growth_chance
				}
				custom_tooltip = "bpm_cell_grows"
			}
		}
	}
}

bpm_increase_or_shrink_cell_level = {
	remove_global_variable = bpm_progressed
	if = {
		limit = {
			NOT = {
				scope:action = {
					is_key_in_variable_map = {
						name = bpm_revcell_progress
						target = prev
					}
				}
			}
		}
		scope:action = {
			update_variable_map_value = {
				name = bpm_revcell_progress
				key = prev
				value = bpm_revcell_base_progress
			}
		}
	}
	bpm_increase_or_shrink_tt = yes
	save_scope_as = state_region
	set_local_variable = {
		name = current_value_p1
		value = {
			value = "scope:action.variable_map(bpm_revcell_progress|scope:state_region)"
			add = 1
		}
	}
	set_local_variable = {
		name = current_value_m1
		value = {
			value = "scope:action.variable_map(bpm_revcell_progress|scope:state_region)"
			add = -1
		}
	}
	if = {
		limit = {
			"scope:action.variable_map(bpm_revcell_progress|scope:state_region)" <= bpm_revcell_progress_min
		}
		scope:action = {
			update_variable_map_value = {
				name = bpm_revcell_progress
				key = prev
				value = bpm_revcell_base_progress
			}
		}
		bpm_reduce_cell_level = yes
	}
	else_if = {
		limit = {
			"scope:action.variable_map(bpm_revcell_progress|scope:state_region)" >= bpm_revcell_progress_max
		}
		scope:action = {
			update_variable_map_value = {
				name = bpm_revcell_progress
				key = prev
				value = bpm_revcell_base_progress
			}
		}
		bpm_increase_cell_level = yes
	}
	else_if = {
		limit = { bpm_revcell_below_front_line = yes }
		random = {
			chance = bpm_revcell_front_line_chance
			scope:action = {
				update_variable_map_value = {
					name = bpm_revcell_progress
					key = prev
					value = local_var:current_value_p1
				}
			}
		}
	}
	else_if = {
		limit = { bpm_revcell_above_front_line = yes }
		random = {
			chance = bpm_revcell_front_line_chance
			scope:action = {
				update_variable_map_value = {
					name = bpm_revcell_progress
					key = prev
					value = local_var:current_value_m1
				}
			}
		}
	}
	else = {
		random = {
			chance = bpm_revcell_front_line_chance
			random_list = {
				0 = {
					modifier = {
						add = bpm_cell_shrinking_chance
					}
					scope:action = {
						update_variable_map_value = {
							name = bpm_revcell_progress
							key = prev
							value = local_var:current_value_m1
						}
					}
				}
				0 = {
					modifier = {
						add = bpm_cell_growth_chance
					}
					scope:action = {
						update_variable_map_value = {
							name = bpm_revcell_progress
							key = prev
							value = local_var:current_value_p1
						}
					}
				}
			}
		}
	}

	remove_local_variable = current_value_m1
	remove_local_variable = current_value_p1
}

bpm_start_civil_war = {
	save_scope_as = origin_country
	if = {
		limit = {
			NOT = {
				any_scope_state = {
					state_region = {
						NOT = { bpm_has_revcell = yes }
					}
				}
			}
		}
		set_local_variable = bpm_civil_war_has_all
	}
	else = {
		remove_local_variable = bpm_civil_war_has_all
	}
	every_state = {
		remove_variable = bpm_state_to_cede
	}

	every_scope_state = {
		limit = {
			state_region = {
				bpm_has_revcell = yes
			}
			OR = {
				NOT = { is_capital = yes }
				NOT = { has_local_variable = bpm_civil_war_has_all }
			}
		}
		set_variable = bpm_state_to_cede
	}
	every_country = {
		remove_variable = bpm_coniq
	}
	random_scope_state = {
		limit = {
			has_variable = bpm_state_to_cede
		}
		save_scope_as = state_to_give
	}
	create_country = {
		origin = scope:origin_country
		state = scope:state_to_give
		on_created = {
			set_variable = bpm_coniq
		}
	}
	every_supportive_interest_group = {
		owner = {
			add_to_variable_list = {
				name = bpmtotransferrev
				target = prev.type
			}
		}
	}
	random_country = {
		limit = {
			has_variable = bpm_coniq
		}
		save_scope_as = new_country
		set_variable = {
			name = bpm_country_revolutionary_action
			value = scope:action
		}
		set_variable = {
			name = bpm_civil_war_origin
			value = scope:origin_country
		}
		every_interest_group = {
			limit = {
				scope:origin_country = {
					is_target_in_variable_list = {
						name = bpmtotransferrev
						target = prev.type
					}
				}
			}
			owner = {
				add_to_variable_list = {
					name = bpm_civil_war_revolutionary_ig
					target = prev
				}
			}
		}
		bpm_make_government_of_national_defence = yes
	}
	clear_variable_list = bpmtotransferrev
	every_scope_state = {
		limit = {
			has_variable = bpm_state_to_cede
		}
		set_state_owner = scope:new_country
	}
	scope:new_country = { create_diplomatic_play = {
		target_country = scope:origin_country
		type = bpm_dp_revolution
		add_war_goal = { holder = scope:origin_country type = bpm_civil_war_preserve target_country = scope:new_country  }
	} }
	scope:new_country = {
		if = {
			limit = {
				exists = var:bpm_civil_war_origin
			}
			every_country = {
				limit = {
					has_variable = bpm_civil_war_origin
					var:bpm_civil_war_origin = prev.var:bpm_civil_war_origin
					NOT = {
						this = prev
					}
				}
				save_scope_as = target_country
				prev = {
					create_diplomatic_play = {
						target_country = scope:target_country
						type = dp_revolution
						escalation = 80
						annex_as_civil_war = yes
					}
				}
			}
		}
	}
}

	}

	var:bpm_holder = { save_scope_as = creator_country }

			}

					name = bpmtotransferrev
					target = prev.type
				}
			}
		}
		play_as = scope:target_country
	}
	scope:target_country = {
		set_variable = {
			name = civil_war_type_var
			value = scope:creator_country.var:civil_war_type_var
		}
		every_interest_group = {
			limit = {
				scope:creator_country = {
					is_target_in_variable_list = {
						name = bpmtotransferrev
						target = prev.type
					}
				}
			}
			owner = {
				add_to_variable_list = {
					name = bpm_civil_war_revolutionary_ig
					target = prev
				}
			}
		}
		annex_as_civil_war = scope:creator_country
		every_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_revolution
			}
			end_play = yes # should this be here??
		}

		bpm_coalition_revolution_victorious = yes
	}
}

bpm_update_grasp_threshold = {
	random_list = {
		1 = {
			set_variable = {
				name = bpm_grasp_front_line
				value = {
					value = bpm_revolt_front_line_centre
					add = -1
					min = 0
				}
			}
		}
		2 = {
			set_variable = {
				name = bpm_grasp_front_line
				value = bpm_revolt_front_line_centre
			}
		}
		1 = {
			set_variable = {
				name = bpm_grasp_front_line
				value = {
					value = bpm_revolt_front_line_centre
					add = 1
					max = 6
				}
			}
		}
	}
	every_in_list = {
		variable = bpm_active_actions
		limit = { var:bpm_action_type ?= flag:bpm_action_revolt }
		set_variable = {
			name = bpm_grasp_threshold
			value = root.var:bpm_grasp_front_line
		}
	}
	bpm_update_grasp_lists = yes
}

bpm_init_missing_front_lines = {
	if = {
		limit = { has_variable = bpm_grasp_front_line }
		every_in_list = {
			variable = bpm_active_actions
			limit = {
				var:bpm_action_type ?= flag:bpm_action_revolt
				NOT = { has_variable = bpm_grasp_threshold }
			}
			set_variable = {
				name = bpm_grasp_threshold
				value = root.var:bpm_grasp_front_line
			}
		}
	}
}

bpm_update_grasp_lists = {
	clear_variable_list = bpm_revolt_actions_gui
	every_in_list = {
		variable = bpm_active_actions
		limit = { var:bpm_action_type ?= flag:bpm_action_revolt }
		root = { add_to_variable_list = { name = bpm_revolt_actions_gui target = prev } }
	}
	if = {
		limit = {
			has_variable = bpm_revolt_selected_action
			NOT = {
				any_in_list = {
					variable = bpm_revolt_actions_gui
					this = root.var:bpm_revolt_selected_action
				}
			}
		}
		remove_variable = bpm_revolt_selected_action
	}
	if = {
		limit = {
			NOT = { has_variable = bpm_revolt_selected_action }
			has_variable_list = bpm_revolt_actions_gui
		}
		random_in_list = {
			variable = bpm_revolt_actions_gui
			save_temporary_scope_as = first_revolt
		}
		set_variable = {
			name = bpm_revolt_selected_action
			value = scope:first_revolt
		}
	}

	clear_variable_list = bpm_grasp_states_1
	clear_variable_list = bpm_grasp_states_2
	clear_variable_list = bpm_grasp_states_3
	clear_variable_list = bpm_grasp_states_4
	clear_variable_list = bpm_grasp_states_5
	clear_variable_list = bpm_revolt_cell_states
	every_scope_state = {
		remove_variable = bpm_grasp_sel
		remove_variable = bpm_grasp_sel_category
		remove_variable = bpm_grasp_sel_level
		remove_variable = bpm_gp_legitimacy
		remove_variable = bpm_gp_turmoil
		remove_variable = bpm_gp_clout
		remove_variable = bpm_gp_neighbour_cells
		remove_variable = bpm_gp_calm_neighbours
		remove_variable = bpm_gp_capital
		remove_variable = bpm_gp_unincorporated
		remove_variable = bpm_gp_repression
		remove_variable = bpm_gp_cells
	}

	if = {
		limit = { has_variable = bpm_revolt_selected_action }
		var:bpm_revolt_selected_action = { save_scope_as = action }
		if = {
			limit = {
				scope:action = { has_variable = bpm_grasp_threshold }
			}
			set_variable = {
				name = bpm_grasp_threshold
				value = scope:action.var:bpm_grasp_threshold
			}
		}
		else = {
			remove_variable = bpm_grasp_threshold
		}
		every_scope_state = {
			set_variable = {
				name = bpm_grasp_sel
				value = bpm_state_grasp
			}
			set_variable = {
				name = bpm_grasp_sel_category
				value = bpm_state_grasp_category
			}
			set_variable = {
				name = bpm_grasp_sel_level
				value = bpm_action_cell_level
			}
			set_variable = {
				name = bpm_gp_legitimacy
				value = bpm_grasp_part_legitimacy
			}
			set_variable = {
				name = bpm_gp_turmoil
				value = bpm_grasp_part_turmoil
			}
			set_variable = {
				name = bpm_gp_clout
				value = bpm_grasp_part_clout
			}
			set_variable = {
				name = bpm_gp_neighbour_cells
				value = bpm_grasp_part_neighbour_cells
			}
			set_variable = {
				name = bpm_gp_calm_neighbours
				value = bpm_grasp_part_calm_neighbours
			}
			set_variable = {
				name = bpm_gp_capital
				value = bpm_grasp_part_capital
			}
			set_variable = {
				name = bpm_gp_unincorporated
				value = bpm_grasp_part_unincorporated
			}
			set_variable = {
				name = bpm_gp_repression
				value = bpm_grasp_part_repression
			}
			set_variable = {
				name = bpm_gp_cells
				value = bpm_grasp_part_cells
			}
		}
		ordered_scope_state = {
			limit = { bpm_state_in_revolt_reach = yes }
			order_by = var:bpm_grasp_sel
			max = 10000
			check_range_bounds = no
			if = {
				limit = { var:bpm_grasp_sel_category = 1 }
				owner = { add_to_variable_list = { name = bpm_grasp_states_1 target = prev } }
			}
			else_if = {
				limit = { var:bpm_grasp_sel_category = 2 }
				owner = { add_to_variable_list = { name = bpm_grasp_states_2 target = prev } }
			}
			else_if = {
				limit = { var:bpm_grasp_sel_category = 3 }
				owner = { add_to_variable_list = { name = bpm_grasp_states_3 target = prev } }
			}
			else_if = {
				limit = { var:bpm_grasp_sel_category = 4 }
				owner = { add_to_variable_list = { name = bpm_grasp_states_4 target = prev } }
			}
			else = {
				owner = { add_to_variable_list = { name = bpm_grasp_states_5 target = prev } }
			}
		}
		ordered_scope_state = {
			limit = { var:bpm_grasp_sel_level > 0 }
			order_by = var:bpm_grasp_sel_level
			max = 10000
			check_range_bounds = no
			owner = { add_to_variable_list = { name = bpm_revolt_cell_states target = prev } }
		}
		clear_saved_scope = action
	}
	else = {
		remove_variable = bpm_grasp_threshold
	}
}
