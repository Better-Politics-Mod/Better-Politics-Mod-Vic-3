bpm_the_docket_cooldown = {
	add_to_variable_list = {
		name = bpm_cooldown_laws
		target = currently_enacting_law.type
		months = 6
	}
}

bpm_update_docket_gui = {
	clear_variable_list = bpm_the_docket
	ordered_law = {
		limit = { bpm_law_is_docket_candidate = yes }
		order_by = bpm_law_docket_score
		prev = { add_to_variable_list = { name = bpm_the_docket target = prev } }
	}
}

bpm_update_docket_outliner_gui = {
	remove_variable = bpm_docket_top
	ordered_law = {
		limit = { bpm_law_is_docket_candidate = yes }
		order_by = bpm_law_docket_score
		position = 0
		check_range_bounds = no
		prev = { set_variable = { name = bpm_docket_top value = prev } }
	}
}

bpm_update_law_enactment_docket = {
	if = {
		limit = {
			enacting_any_law = no
		}
		ordered_law = {
			limit = { bpm_law_is_docket_candidate = yes }
			order_by = bpm_law_docket_score
			position = 0
			check_range_bounds = no
			save_temporary_scope_as = next_law
			if = {
				limit = {
					bpm_law_executive_would_edict = yes
				}
				owner = {
					bpm_start_enactment_as_edict = {
						lawtype = scope:next_law.type
					}
				}
			}
			else = {
				owner = { start_enactment = scope:next_law.type }
			}
		}
	}
}
