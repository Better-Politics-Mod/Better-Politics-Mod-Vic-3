law_ceremonial_hos = {
	group = lawgroup_executive_principles
	
	icon = "gfx/interface/icons/law_icons/answers.dds"
	
	progressiveness = 50
	
	modifier = {
		country_bpm_hog_veto = yes
		country_bpm_hog_selected_by_legislature = yes
	}

	is_visible = {
		OR = {
			NOT = { c:JAP ?= this }
			NOR = {
				NOT = { has_variable = japan_restoration_complete }
				has_law = law_type:law_bakufu
			}
		}
	}

	can_enact = {
	}

	on_activate = {
		if = {
			limit = {
				has_law = law_type:law_presidential_republic
			}
			activate_law = law_type:law_parliamentary_republic
		}
		if = {
			limit = {has_game_started = yes}
			bpm_end_regency = yes
		}
	}

	on_deactivate = {
		hidden_effect = {
			if = {
				limit = {NOT = {has_law_or_variant = law_type:law_ceremonial_hos}}
				bpm_end_acting_hog = yes
			}
		}
		if = {
			limit = {
				has_law = law_type:law_monarchy
				ruler = {
					is_adult = no
					NOT = { has_variable = regency_complete }
				}
			}
			trigger_event = {
				id = regency.1
				popup = yes
			}
		}
	}

	disallowing_laws = {
	}
}

REPLACE:law_bakufu = {
	group = lawgroup_executive_principles
	parent = law_ceremonial_hos

	icon = "gfx/interface/icons/law_icons/bakufu.dds"

	progressiveness = -100

	unlocking_laws = {
		law_monarchy
		law_social_monarchy
	}

	is_visible = {
		country_has_primary_culture = cu:japanese
		OR = {
			has_law = law_type:law_bakufu
			NOT = {has_variable = japan_restoration_complete}
		}
	}

	on_activate = {
		hidden_effect = {
			country_assess_daimyo_alignments = yes
		}
		if = {
			limit = {has_game_started = yes}
			bpm_end_regency = yes
		}
	}

	on_deactivate = {
		if = {
			limit = {
				country_has_monarchy_law = yes
				country_has_non_monarchy_law = no # To handle an execution order issue
			}
			hidden_effect = {
				country_assess_daimyo_alignments = yes
			}
		}
		hidden_effect = {
			if = {
				limit = {NOT = {has_law_or_variant = law_type:law_ceremonial_hos}}
				bpm_end_acting_hog = yes
			}
		}
		if = {
			limit = {
				has_law = law_type:law_monarchy
				ruler = {
					is_adult = no
					NOT = { has_variable = regency_complete }
				}
			}
			trigger_event = {
				id = regency.1
				popup = yes
			}
		}
	}

	modifier = {
		country_aristocrats_pol_str_mult = 0.25
		country_officers_pol_str_mult = 0.5
		country_soldiers_pol_str_mult = 0.25
		country_bpm_hog_veto = yes
		country_bpm_hog_selected_by_legislature = yes
	}

	can_impose = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
	}

	ai_impose_chance = {
		value = 0
	}
}
