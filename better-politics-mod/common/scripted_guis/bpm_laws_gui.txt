bpm_repeatable_law_check_gui = {
	scope = country
	saved_scopes = {
		law_type
	}

	is_shown = {
		is_target_in_global_variable_list = {
            name = bpm_laws_repeatable_category
            target = scope:law_type
        }
		OR = {
			NOT = { scope:law_type = { is_same_law_group_as = law_type:law_repeatable_special_none } }
			has_variable = bpm_enacting_special_law
		}
	}
}
