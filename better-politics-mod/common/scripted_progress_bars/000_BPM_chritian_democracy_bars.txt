je_bpm_christian_democrat_independence_progress_bar = {
	name = "je_bpm_christdem_independence_progress_bar"
	desc = "je_bpm_christdem_independence_progress_bar_desc"

	start_value = 0
	min_value = 0

			limit = {
				owner = {
					is_player = no
					country_has_state_religion = rel:catholic
				}
			}
			add = 0.066
		}

		##### Ironically the more intertwined the church and the state are the more the church wants to stop the laymen.
		progress_condition = {
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					has_law_or_variant = law_type:law_state_religion
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_STATE_RELIGION"
				value = 0.05
			}
			else = {
				limit = {
					owner = {
						country_has_state_religion = rel:catholic
						OR = {
							has_law = law_type:law_freedom_of_conscience
							has_law_or_variant = law_type:law_total_separation
						}
					}
				}
				add = {
					desc = "BPM_INDEPENDENCE_FROM_SEPARATION_OF_CHURCH"
					value = 0.033
				}
				else = {
					limit = {
						owner = {
							country_has_state_religion = rel:catholic
							has_law = law_type:law_state_atheism
						}
					}
					add = {
						desc = "BPM_INDEPENDENCE_FROM_STATE_ATHEISM"
						value = 0.025
					}
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					any_interest_group = { is_interest_group_type = ig_liberals }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_LIBERALS"
				value = owner.ig:ig_liberals.ig_clout
				multiply = 1.1
				divide = 2.5
			}
		}

		progress_condition = {
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					any_interest_group = { is_interest_group_type = ig_radicals }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_RADICALS"
				value = owner.ig:ig_radicals.ig_clout
				multiply = 1.33
				divide = 2
			}
		}

		progress_condition = { # They're practically off the leash, and are exploding in popularity.
			limit = {
				owner = { country_has_state_religion = rel:catholic }
				has_global_variable = global_rerum_novarum_issued_var
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_RERUM_NOVARUM"
				value = 0.2
			}
		}

		progress_condition = {
			limit = {
				owner = { country_has_state_religion = rel:catholic }
				has_global_variable = global_variable_graves_de_communi_re
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_GRAVES_DE_COMMUNI_RE"
				value = -0.1
			}
		}

		##### Political movements
		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_labor }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_LABOR_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:bpm_labor_movement.political_movement_support
					multiply = 0.15
				}
				else = {
					value = 0.02
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_liberal }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_LIBERAL_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:bpm_liberal_movement.political_movement_support
					multiply = 0.25
				}
				else = {
					value = 0.02
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_radical }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_RADICAL_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:bpm_radical_movement.political_movement_support
					multiply = 0.33
				}
				else = {
					value = 0.01
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_modernizer }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_MODERNIZER_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:bpm_modernizer_movement.political_movement_support
					multiply = 0.5
				}
				else = {
					value = 0.004
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_socialist }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_SOCIALIST_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:bpm_socialist_movement.political_movement_support
					multiply = 0.44
				}
				else = {
					value = 0.009
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_communist }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_COMMUNIST_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:movement_communist.political_movement_support
					multiply = 0.25
				}
				else = {
					value = 0.011
				}
			}
		}

		progress_condition = {
			limit = {
				owner = {
					any_political_movement = { is_political_movement_type = movement_anarchist }
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_ANARCHIST_MOVEMENT"
				if = {
					limit = { owner = { is_player = yes } }
					value = owner.var:movement_anarchist.political_movement_support
					multiply = 0.25
				}
				else = {
					value = 0.005
				}
			}
		}

		##### Technologies
		progress_condition = {
			limit = {
				owner = { has_technology_researched = labor_movement }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_TECH_LABOR_MOVEMENT"
				value = 0.001
			}
		}

		progress_condition = {
			limit = {
				owner = { has_technology_researched = corporatism }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_TECH_CORPORATISM"
				value = 0.003
			}
		}

		progress_condition = {
			limit = {
				owner = { has_technology_researched = socialism }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_TECH_SOCIALISM"
				value = 0.01
			}
		}

		progress_condition = {
			limit = {
				owner = { has_technology_researched = political_agitation }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_TECH_POLITICAL_AGITATION"
				value = 0.005
			}
		}

		progress_condition = { ##### The church's strength translates to slower laymen independence
			limit = {
				owner = { country_has_state_religion = rel:catholic }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_DEVOUT"
				value = owner.ig:ig_devout.ig_clout
				multiply = -5
			}
		}

		progress_condition = { ##### Christian Democrats/Catholic Interests are rising, they got to break away from the church.
			limit = {
				owner = { country_has_state_religion = rel:catholic }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_CHRISTIAN_DEMOCRATS"
				value = owner.ig:ig_christian_democrats.ig_clout
			}
		}

		##### Strong moralist can effectively combat any brave laymen.
		progress_condition = { ### Conservatives hate the gallant laymen!
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					any_interest_group = {
						is_interest_group_type = ig_conservatives
						has_ideology = ideology:ideology_rel_moralist
					}
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_MORALIST_CONSERVATIVES"
				value = owner.ig:ig_conservatives.ig_clout
				multiply = -2.5
			}
		}

		progress_condition = { ### Reactionaries even more!
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					any_interest_group = {
						is_interest_group_type = ig_reactionaries
						has_ideology = ideology:ideology_rel_radical_moralist
					}
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_MORALIST_REACTIONARIES"
				value = owner.ig:ig_reactionaries.ig_clout
				multiply = -2.5
			}
		}

		progress_condition = { ### National Liberals dislike laborers
			limit = {
				owner = {
					country_has_state_religion = rel:catholic
					any_interest_group = {
						is_interest_group_type = ig_national_liberals
						has_ideology = ideology:ideology_rel_moralist
					}
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_MORALIST_NATIONAL_LIBERALS"
				value = owner.ig:ig_national_liberals.ig_clout
				multiply = -1.33
			}
		}

		progress_condition = {
			limit = {
				owner = { exists = ig:ig_trade_unions }
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_TRADE_UNIONS"
				value = owner.ig:ig_trade_unions.ig_clout
				multiply = 0.5
			}
		}

		progress_condition = {
			limit = {
				owner = {
					exists = ig:ig_christian_democrats
					has_variable = christian_democrats_ideology_change
				}
			}
			add = {
				desc = "BPM_INDEPENDENCE_FROM_IDEOLOGY_CHANGES"
				value = owner.var:christian_democrats_ideology_change
				multiply = owner.ig:ig_christian_democrats.ig_clout
				divide = 2.5
			}
		}
	}
}
