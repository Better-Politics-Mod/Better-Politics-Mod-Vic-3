bpm_make_into_agitator = {
	clicksound = "event:/SFX/DLC/1.3_ip1/UI/character_invite"

	potential = {
		is_in_exile_pool = no
		is_in_void = no
		owner ?= { is_player = yes }
		is_ruler_of_own_country = no
		is_heir_of_own_country = no
		has_role_of_type = politician
		NOT = { has_role_of_type = agitator }
		interest_group = {
			OR = {
				NOT = { has_ideology = ideology:ideology_nonexistent }
				NOT = {
					is_interest_group_type = ig_christian_democrats
				}
			}
		}
		NOT = { has_variable = bpm_untouchable_char }
		NOR = {
			has_role = bpm_character_role_hog
			has_role = bpm_character_role_protege
			has_role = bpm_character_role_regent
            has_role = bpm_character_role_acting_hog
		}
	}

	show_requirements = yes

	possible = {
		owner ?= { is_player = yes }
		scope:actor = { empty_agitator_slots >= 1 }
		interest_group = {
			bpm_ig_is_ideological = yes
			is_insurrectionary = no
		}
		NOT = { has_ideology = ideology:ideology_moderate }
	}

	effect = {
		#hidden_effect = {
			if = {
				limit = {
					NOT = { has_role_of_type = agitator }
				}
				add_character_role = agitator
			}
			if = {
				limit = {
					has_role_of_type = general
				}
				remove_character_role = general
			}
			if = {
				limit = {
					has_role_of_type = admiral
				}
				remove_character_role = admiral
			}
			if = {
				limit = {
					has_role_of_type = politician
				}
				set_as_interest_group_leader = no
				# remove_as_interest_group_leader = yes
				remove_character_role = politician
			}
			hidden_effect = {
				exile_character = yes
				transfer_character = scope:actor
			}
		#}
	}

	cooldown = { days = normal_modifier_time }
	show_confirmation_box = yes

	ai_chance = {
		value = -50
	}
}

bpm_appoint_pm = {
	clicksound = "event:/SFX/DLC/1.3_ip1/UI/character_invite"
	icon = "gfx/interface/icons/character_role_icons/bpm_minister.dds"

	potential = {
		exists = owner
		exists = interest_group

		owner ?= {
			OR = {
				bpm_is_hosapp = yes
				modifier:country_bpm_hog_top_advice = yes
			}
			# has_variable = bpm_has_cabinet
			NOT = { var:bpm_headofgov ?= { has_variable = bpm_untouchable_char } }
		}

		bpm_hog_valid_triggers = yes
		NOT = {has_role = bpm_character_role_hog}

		#NOT = { has_variable = bpm_untouchable_char }
	}

	show_requirements = yes

	possible = {
		bpm_char_can_be_hog = yes
	}


	effect = {
		bpm_appoint_as_hog = yes
	}

	show_confirmation_box = no
}

		}
	}

	show_requirements = yes

	possible = { 
		
	}

		owner = {
			bpm_clear_hog = yes
		}
	}

	show_confirmation_box = no
