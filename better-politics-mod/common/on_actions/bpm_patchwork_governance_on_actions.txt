on_monthly_pulse_state = {
	on_actions = {
		bpm_patchwork_governance_state_monthly_oa
	}
}

on_monthly_pulse_country = {
	on_actions = {
		bpm_patchwork_governance_country_monthly_oa
	}
}

on_game_started = {
	on_actions = {
		bpm_patchwork_governance_game_start_oa
	}
}

# this = state
bpm_patchwork_governance_state_monthly_oa = {
	effect = {
		bpm_patchwork_governance_state_monthly = yes
	}
}

# this = country
bpm_patchwork_governance_country_monthly_oa = {
	effect = {
		bpm_patchwork_governance_country_monthly = yes
	}
}

bpm_patchwork_governance_game_start_oa = {
	effect = {
		every_country = {
			bpm_patchwork_governance_game_start = yes
		}
	}
}
