bpm_update_consumption_tax_anger = { # this = country
	save_scope_as = bpm_ctax_country
	bpm_count_taxed_goods_in_category = { CATEGORY = staple_consumer_goods }
	bpm_count_taxed_goods_in_category = { CATEGORY = luxury_consumer_goods }
	bpm_count_taxed_goods_in_category = { CATEGORY = luxury_food_and_drink }
	bpm_count_taxed_goods_in_category = { CATEGORY = cash_crops }
	bpm_count_taxed_goods_in_category = { CATEGORY = local_services }
	bpm_count_taxed_goods_in_category = { CATEGORY = primary_industrial_resources }
	bpm_count_taxed_goods_in_category = { CATEGORY = manufactured_industrial_goods }
	bpm_count_taxed_goods_in_category = { CATEGORY = military_goods }
	bpm_count_taxed_goods_in_category = { CATEGORY = shipping_goods }
	every_interest_group = {
		remove_modifier = bpm_ig_consumption_tax_anger
		if = {
			limit = { bpm_ig_consumption_tax_burden > 0 }
			add_modifier = {
				name = bpm_ig_consumption_tax_anger
				multiplier = bpm_ig_consumption_tax_burden
			}
		}
	}
}

bpm_count_taxed_goods_in_category = { # this = country
	set_variable = {
		name = bpm_ctax_count_$CATEGORY$
		value = 0
	}
	if = {
		limit = { num_taxed_goods > 0 }
		every_in_global_list = {
			variable = bpm_ctax_$CATEGORY$
			limit = {
				scope:bpm_ctax_country = { has_consumption_tax = prev }
			}
			scope:bpm_ctax_country = {
				change_variable = {
					name = bpm_ctax_count_$CATEGORY$
					add = 1
				}
			}
		}
	}
}
