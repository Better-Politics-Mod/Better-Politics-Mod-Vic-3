REPLACE:ig_petty_bourgeoisie = {
	color = hsv{ 0.59 0.12 0.6 }
	texture = "gfx/interface/icons/ig_icons/petty_bourgeoisie.dds"
	layer = "revolution_dynamic_petty_bourgeoisie"
	index = 5

	ideologies = {
		# Vanilla
		ideology_meritocratic
		ideology_patriotic
	}
	
	character_ideologies = {
		ideology_moderate
		ideology_traditionalist
	}

	enable = {
		always = yes
	}	
	
	on_enable = {
		bpm_update_name_petty_bourgeoisie = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				assamese avadhi baluchi bengali bihari gujarati kannada malayalam marathi oriya panjabi pathan rajput sindi tamil telegu hindustani bundeli pahari gondi bageli chhattisgarhi deccani
			}
		}
		rule = {
			trigger = {
				c:ZAN ?= this
				any_primary_culture = {
					cu:bedouin = this
				}
			}
			cultures = {
				swahili gujarati
			}
		}
	}

	on_character_ig_membership = {
		if = {
			limit = {
				culture = cu:gujarati
				owner = {
					c:ZAN ?= this
					any_primary_culture = {
						cu:bedouin = this
					}
				}
			}
			change_character_religion = rel:shiite
		}
	}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				NOT = {
					is_country_type = decentralized
				}
			}
		}
		OR = {
			is_pop_type = clerks
			is_pop_type = bureaucrats
			is_pop_type = aristocrats
		}
	}	
	
	pop_weight = {
		value = 0

		if = {
			limit = { is_pop_type = clerks }
			add = {
				desc = "POP_CLERKS"
				value = 20
				if = {
					limit = { workplace = { is_government_funded = yes } }
					add = 50
				}
			}
		}
		else_if = {
			limit = { is_pop_type = bureaucrats }
			add = {
				desc = "POP_BUREAUCRATS"
				value = 100
			}
		}
		else_if = {
			limit = {
				is_pop_type = aristocrats
				workplace = { is_government_funded = yes }
			}
			add = {
				desc = "POP_ARISTOCRATS"
				value = 75
			}
		}

		multiply = ig_attraction_to_interests_from_stability	
		multiply = {
			desc = "MULTIPLIER"	
			value = 1
	
			add = {
				scope:interest_group = {
					leader = {
						value = popularity
						multiply = 0.01
						add = 1
						max = 2.0
						min = 0.75
					}
				}
			}
			add = {
				value = 0
				if = {
					limit = {
						scope:interest_group = {
							is_in_government = yes
						}
					}
					add = {
						desc = "IN_GOVERNMENT_ATTRACTION"
						value = 1
						add = scope:interest_group.modifier:interest_group_in_government_attraction_mult
						min = 0
					}
				}
			}
		}
	}
	
	monarch_weight = {
		owner.ig:ig_petty_bourgeoisie = {
			value = bpm_ig_monarch_weight
		}
	}

	agitator_weight = {
		owner.ig:ig_petty_bourgeoisie = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_petty_bourgeoisie = {
			value = bpm_commander_weight
		}
	}

	executive_weight = {
		value = 0.2
		if = {
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_cooperative_ownership
						has_law_or_variant = law_type:law_command_economy
					}
				}
			}
			value = 5.0
		}
	}
	
	noble_chance = {
		value = 0.5
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
