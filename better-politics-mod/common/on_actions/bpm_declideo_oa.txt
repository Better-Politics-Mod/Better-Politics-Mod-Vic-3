on_game_started = {
    on_actions = {
        bpm_on_game_started_declideo
    }
}

on_law_activated = {
    on_actions = {
        bpm_on_law_activated_declideo
        bpm_update_entrenchment
    }
}

bpm_on_game_started_declideo = {
    effect = {
        every_country = {
            bpm_update_ideology_all_igs = yes
        }
    }
}

bpm_on_law_activated_declideo = {
    effect = {
        if = {
            limit = { has_game_started = yes }
            bpm_make_newly_enacted = yes
        }
    }
}

bpm_update_entrenchment = {
    effect = {
        owner = {
            bpm_update_entrenched_laws = yes
        }
    }
}
