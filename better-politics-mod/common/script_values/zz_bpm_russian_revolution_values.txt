REPLACE:bolshevik_value = {
	value = 0
	every_political_movement = {
		limit = {
			is_political_movement_type = movement_communist
		}
		add = {
			desc = "socialist_faction_movement_desc"
			value = this.political_movement_support
			multiply = 200
			max = 50
		}
	}
	every_political_movement = {
		limit = {
			is_political_movement_type = movement_labor
		}
		add = {
			desc = "socialist_faction_labor_desc"
			value = this.political_movement_support
			multiply = ROOT.var:bolshevik_labor
			max = 50
		}
	}
	if = {
		limit = {
			exists = var:bolshevik_leader
		}
		add = {
			desc = "socialist_faction_leader_prominence_desc"
			value = var:bolshevik_leader.prominence
			multiply = 0.5
			max = 25
		}
		add = {
			desc = "socialist_faction_leader_popularity_desc"
			value = var:bolshevik_leader.popularity
			multiply = 0.5
			max = 25
		}
	}
	ig:ig_revolutionist_socialists = {
		add = {
			desc = "socialist_faction_ig_clout_desc"
			value = this.interest_group_population_percentage
			multiply = 200
			max = 50
		}
	}
}
