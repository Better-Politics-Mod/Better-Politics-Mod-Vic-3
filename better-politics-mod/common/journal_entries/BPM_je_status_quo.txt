je_bpm_multicultural_upgrade = {
	icon = "gfx/interface/icons/law_icons/multicultural.dds"

	group = je_group_internal_affairs
	
	scripted_progress_bar = je_bpm_multicultural_upgrade_progress_bar

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = "bpm_je_progress_complete"
				scope:journal_entry = {
					"scripted_bar_progress(je_bpm_multicultural_upgrade_progress_bar)" >= 100
				}
			}
		}
		effect = {
			activate_law = law_type:law_multicultural_upgraded
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				has_law = law_type:law_multicultural
			}
		}
	}

	weight = 200

	should_be_pinned_by_default_uninvolved_or_context = yes
}
