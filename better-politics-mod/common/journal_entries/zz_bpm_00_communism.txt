REPLACE:je_communism_1 = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		NOT = { has_variable = communism_1_done }
		OR = {
			has_technology_researched = socialism
			has_technology_researched = anarchism
		}
	}

	scripted_progress_bar = communism_1_progress_bar

	possible = {
		any_interest_group = {
			bpm_ig_is_party_socialist = yes
			bpm_ig_is_marginal = no
		}
		ig:ig_trade_unions ?= {
			bpm_ig_is_marginal = no
		}
	}

	immediate = {
		root = {
			save_scope_as = root_country
		}
		set_communism_vars = yes
		trigger_event = { id = communism.10 days = 1 popup = yes }
	}

	on_weekly_pulse = {
		effect = {
			calculate_communism_progress = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			5 = communism.7
		}
		effect = {
			calculate_communism_progress = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = communism_progress_bar_at_full_tt
					scope:journal_entry = {
						"scripted_bar_progress(communism_1_progress_bar)" >= 100
					}
				}
				has_law_or_variant = law_type:law_council_republic
			}
		}
		effect = {
			set_variable = communism_1_done
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_council_republic
					}
				}
				trigger_event = { id = communism.1 popup = yes }
			}
			else = { # Mirrors effects of communism.1, skips je_communism_2
				if = {
					limit = {
						any_interest_group = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
					}
					random_interest_group = {
						limit = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
						set_variable = communist_ig_var
					}
				}
				else = {
					ig:ig_trade_unions ?= {
						set_variable = communist_ig_var
					}
				}
				trigger_event = { id = communism.2 popup = yes }
			}
			custom_tooltip = communism_je_1_fail_tt
		}
	}

	timeout = 1095

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			trigger_event = { id = communism.11 popup = yes }
			custom_tooltip = communism_je_1_complete_tt
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
}
