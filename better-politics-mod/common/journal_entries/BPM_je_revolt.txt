je_bpm_revolt_ember = {
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	group = je_group_bpm_upheaval

	is_shown_when_inactive = {
		always = yes
	}

	widget = {
		gui = "gui/journal_entry_widgets/bpm_revolt_widgets.gui"
		name = "widget_je_bpm_revolt_grasp"
		container = "custom_widget_container_2" # directly below the status
	}

	immediate = {
		bpm_update_grasp_threshold = yes
	}

	on_monthly_pulse = {
		effect = {
			bpm_update_grasp_threshold = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			bpm_init_missing_front_lines = yes
			bpm_update_grasp_lists = yes
		}
	}

	can_deactivate = yes
	possible = {
		bpm_has_revolt_action = yes
	}

	map_tooltip = {
		if = {
			limit = { var:bpm_grasp_sel_level ?= 4 }
			custom_tooltip_no_bullet = "je_bpm_revolt_ember_tt_lvl4"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 3 }
			custom_tooltip_no_bullet = "je_bpm_revolt_ember_tt_lvl3"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 2 }
			custom_tooltip_no_bullet = "je_bpm_revolt_ember_tt_lvl2"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 1 }
			custom_tooltip_no_bullet = "je_bpm_revolt_ember_tt_lvl1"
		}
		else = {
			custom_tooltip_no_bullet = "je_bpm_revolt_ember_tt_none"
		}
	}

	map_color = {
		if = {
			limit = { var:bpm_grasp_sel_level ?= 4 }
			value = "bpm_revmap_cell_4"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 3 }
			value = "bpm_revmap_cell_3"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 2 }
			value = "bpm_revmap_cell_2"
		}
		else_if = {
			limit = { var:bpm_grasp_sel_level ?= 1 }
			value = "bpm_revmap_cell_1"
		}
		else = {
			value = "bpm_revmap_own_state"
		}
	}

	outcome = {
		flags = { hidden }
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			always = no
		}
	}

	transferable = no
	can_revolution_inherit = no

	weight = 10
	should_be_pinned_by_default_uninvolved_or_context = yes
}
