GLOBAL = {
    bpm_setup_global_vars = yes
    every_country = {
        # colonies need to be independent for the duration of character history or it will crash the game since we enable use_overlord_ruler for them
        # they are remade colonies in government_setup
        # they have to remain colonies earlier in history for government types to init properly
        bpm_setup_colonial_laws = yes
        if = {
            limit = {is_subject_type = subject_type_colony}
            set_variable = {
                name = colonial_overlord
                value = overlord
            }
            make_independent = yes
        }
        else_if = {
            limit = {is_subject_type = subject_type_dominion}
            set_variable = {
                name = dominion_overlord
                value = overlord
            }
            make_independent = yes
        }
        every_country = {
        set_variable = {
            name = chud
            value = bureaucracy
        }
    }
    }
}
