# Game start ideology setup lives in bpm_game_start_country_effect, so that it is
# guaranteed to run before the ruling IG and the starting coalition are decided

on_law_activated = {
    on_actions = {
        bpm_on_law_activated_declideo
        bpm_update_entrenchment
    }
}

on_half_yearly_pulse_country = {
    on_actions = {
        bpm_half_yearly_declideo
    }
}

bpm_half_yearly_declideo = {
    effect = {
        if = {
            limit = {
                is_player = no
                country_rank < rank_value:great_power
            }
            bpm_update_ideology_all_igs = yes
        }
    }
}

bpm_on_law_activated_declideo = {
    effect = {
        if = {
            limit = { has_game_started = yes }
            bpm_make_newly_enacted = yes
        }
    }
}

bpm_update_entrenchment = {
    effect = {
        owner = {
            bpm_update_entrenched_laws = yes
            if = {
                limit = { has_game_started = yes }
                bpm_update_ideology_all_igs = yes
            }
        }
    }
}
