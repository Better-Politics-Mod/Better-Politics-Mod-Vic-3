COUNTRIES = {
    every_country = {	
		bpm_setup_country = yes

		## EXECUTIVE DEFAULT
		if = {
			limit = {
				has_law = law_type:law_parliamentary_republic
			}
			activate_law = law_type:law_executive_institutional
		}
		else_if = {
			limit = {
				has_law_or_variant = law_type:law_presidential_republic
				NOR = {
					has_law_or_variant = law_type:law_autocracy
					has_law_or_variant = law_type:law_oligarchy
				}
			}
			activate_law = law_type:law_executive_popular
		}
		else_if = {
			limit = {
				bpm_country_is_pre_constitutional = yes
			}
			activate_law = law_type:law_executive_traditional
		}
		else = {
			activate_law = law_type:law_executive_institutional
		}

		## LABOR DEFAULTS
		activate_law = law_type:law_no_labour_law

		## OFFICER DEFAULTS
		if = {
			limit = {
				OR = {
					has_law = law_type:law_chiefdom
					has_law = law_type:law_monarchy
					has_law = law_type:law_theocracy
				}
			}
			activate_law = law_type:law_noble_officers
		}
		else = {
			activate_law = law_type:law_purchased_officers
		}

		## CENTRALIZATION DEFAULTS
		if = {
			limit = {
				OR = {
					country_has_primary_culture = cu:north_german
					country_has_primary_culture = cu:south_german
					country_has_primary_culture = cu:north_italian
				}
			}
			activate_law = law_type:law_unitary
			set_institution_investment_level = {
				institution = institution_centralization
				level = 1
			}
		}
		else_if = {
			limit = {
				is_in_geographic_region = geographic_region_new_world
			}
			activate_law = law_type:law_federated
			set_institution_investment_level = {
				institution = institution_centralization
				level = 2
			}
		}
		else = {
			activate_law = law_type:law_ad_hoc_centralization
		}

		## SPEECH DEFAULTS
		if = {
			limit = {
				is_country_type = recognized
			}
			activate_law = law_type:law_censorship
		}
		else = {
			activate_law = law_type:law_traditional_ostracism
		}
	}
}
