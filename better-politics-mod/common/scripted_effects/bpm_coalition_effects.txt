bpm_coalition_and_actions_init = {
	add_to_global_variable_list = {
		name = bpm_ig_strategies
		target = flag:bpm_strategy_default
	}
	add_to_global_variable_list = {
		name = bpm_coalition_actions
		target = flag:bpm_action_revolt
	}
}

bpm_coalition_and_actions = {
	save_scope_as = country
	# == Coalition and Actions
	# First each IG should decide on a strategy & an action
	every_activated_interest_group = {
		save_scope_as = ig
		bpm_ig_decide_strategy = yes
		bpm_ig_select_action = yes
		clear_saved_scope = ig
	}

	clear_saved_scope = country
}

bpm_ig_decide_strategy = {
	ordered_in_global_list = {
		variable = bpm_ig_strategies
		position = 0
		order_by = bpm_strategy_attraction

		save_scope_as = selected_strategy
	}

	set_variable = {
		name = bpm_ig_strategy
		value = scope:selected_strategy
	}

	clear_saved_scope = selected_strategy
}

bpm_ig_select_action = {
	# get what action type this IG is most attracted to
	ordered_in_global_list = {
		limit = {
			bpm_ig_action_attraction > 0
		}
		variable = bpm_coalition_actions
		position = 0
		order_by = bpm_ig_action_attraction

		save_scope_as = selected_action_type
	}

	# if the IG wants to do the same action as currently
	# we just do nothing
	# if the IG wants to do a different action as currently
	# we do the process for the IG doing that action
	# if the IG doesn't want to do any action, we do nothing

	if = {
		limit = {
			# if the IG is involved it must be in an action of a different type than the one it wants to do
			# because otherwise we should just do nothing
			exists = scope:selected_action_type
			scope:country = { 
				any_in_list = {
					variable = bpm_active_actions
					count = all
					OR = {
						NOT = { is_target_in_variable_list = {
							name = bpm_actively_involved
							target = scope:ig
						} }
						NOT = { var:bpm_action_type ?= scope:selected_action_type }
					}
				} 
			}
		}

		scope:country = {
			create_container = {
				parent = scope:country
				tags = { bpm_action }
				save_scope_as = created_action
			}
		}

		scope:created_action = {
			add_to_variable_list = {
				name = bpm_actively_involved
				target = scope:ig
			}
			set_variable = {
				name = bpm_action_type
				value = scope:selected_action_type
			}
		}
		
		scope:ig = {
			set_variable = {
				name = bpm_action
				value = scope:created_action
			}
		}

		scope:country = {
			add_to_variable_list = {
				name = bpm_active_actions
				target = scope:created_action
			}
		}

		scope:created_action = {
			save_scope_as = current_action
		}

		clear_saved_scope = created_action
		clear_saved_scope = selected_action_type
	}

	# An IG can only be actively involved in one action at a time
	every_in_list = {
		variable = bpm_active_actions
		limit = {
			NOT = {
				scope:current_action ?= this
			}
			is_target_in_variable_list = {
				name = bpm_actively_involved
				target = scope:ig
			}
		}
		remove_list_variable = {
			name = bpm_actively_involved
			target = scope:ig
		}
	}

	clear_saved_scope = current_action
}




bpm_actions_weekly = {
	save_scope_as = country

	every_in_list = {
		variable = bpm_active_actions
		save_scope_as = action
		bpm_do_action = yes
		clear_saved_scope = action
	}

	clear_saved_scope = country
}
# scope:country
# scope:action
bpm_do_action = {
	bpm_delete_action_if_needed = yes
	if = {
		limit = {
			exists = scope:action
		}
		if = {
			limit = {
				scope:action.var:bpm_action_type ?= flag:bpm_action_revolt
			}
			bpm_action_revolt_do = yes
		}
	}
}

bpm_delete_action_if_needed = {
	if = {
		limit = {
			scope:action = {
				variable_list_size = {
					name = bpm_actively_involved
					value = 0
				}
			}
		}
		if = {
			limit = {
				var:bpm_action_type ?= flag:bpm_action_revolt
			}
			bpm_clear_revolt_stuff = yes
		}
		remove_list_variable = {
			name = bpm_active_actions
			target = scope:action
		}
		scope:action = {
			destroy_container = yes
		}
	}
}

bpm_clear_revolt_stuff = {
	scope:country = {
		remove_from_variable_map = {
			name = bpm_revolt_progress
			key = scope:action
		}
		remove_list_variable = {
			name = bpm_revcell_recently_subsided
			target = scope:action
		}
		every_scope_state = {
			remove_from_variable_map = {
				name = bpm_revcells
				key = scope:action
			}
			remove_from_variable_map = {
				name = bpm_revcell_progress
				key = scope:action
			}
			remove_list_variable = {
				name = bpm_revcell_recently_subsided
				target = scope:action
			}
		}
	}	
}
