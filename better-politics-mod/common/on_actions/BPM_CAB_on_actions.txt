on_monthly_pulse_country = {
    on_actions = {
        bpm_reload_hog
    }
}

on_government_reformed = {
	on_actions = {
		bpm_reload_hog
	}
}

on_election_campaign_start = {
	on_actions = {
		bpm_reload_hog
	}
}

on_election_campaign_end = {
	on_actions = {
		bpm_reload_hog
	}
}

on_character_death = {
    on_actions = {
        bpm_reload_hog
        bpm_update_chars
    }
}

bpm_update_chars = {
    effect = {
        owner = {
            bpm_update_regent = yes
            bpm_update_acting_hog = yes
        }
    }
}

bpm_reload_hog = {
    effect = {
        owner = {
            bpm_update_hog = yes
        }
    }
}
