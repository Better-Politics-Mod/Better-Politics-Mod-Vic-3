ig_fascists = {
	color = hsv{ 0.10 1 0.5 }
	texture = "gfx/interface/icons/ig_icons/fascists.dds"
	layer = "revolution_dynamic_petty_bourgeoisie"
	index = 11
	
	ideologies = {
		ideology_nonexistent
	}

	character_ideologies = {
		ideology_center_leader
		ideology_jingoist_leader
		ideology_ethno_nationalist
	}

	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_fascists = yes
		bpm_update_name_fascists = yes
	}
	on_disable = {}
	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner ?= {
				NOT = { is_country_type = decentralized }
				ig:ig_fascists ?= { bpm_interest_group_is_activated = yes }
			}
		}
		pop_acceptance >= acceptance_status_4
	}
	
	pop_weight = {
		value = 1
	}
	
	monarch_weight = {
		owner.ig:ig_fascists = {
			value = bpm_ig_monarch_weight
		}
	}
	
	agitator_weight = {
		owner.ig:ig_fascists = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_fascists = {
			value = bpm_commander_weight
		}
	}

	executive_weight = {
		value = 0.2
		if ={
			limit = {
				owner.ig:ig_fascists = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.3
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.25
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
