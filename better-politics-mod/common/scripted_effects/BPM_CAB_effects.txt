bpm_update_hog = {
    clear_saved_scope = ruling_ig

    every_scope_character = {
        limit = { has_role = bpm_character_role_hog }
        remove_character_role = bpm_character_role_hog
    }

    every_interest_group = {
        limit = {
            OR = {
                NOT = {
                    owner.var:bpm_headofgov ?= this
                }
                owner = { modifier:country_government_formation_type_institutional = no }
            }
        }
        remove_variable = bpm_institutional_ruling_ig
    }

    ## In popular systems, the elected ruling IG remains the ruling IG until a next election
    ## Any way to dethrone this ruling IG involves new elections 
    ## or switching to an institutional system
    if = {
        limit = { 
            modifier:country_government_formation_type_popular = yes
        }
        if = {
            limit = {
                country_has_voting_franchise = yes
                bpm_country_is_republic = yes
                exists = ruler.interest_group
                NOT = {
                    get_ruler_for:presidential_elective.interest_group ?= ruler.interest_group
                }
            }
            ruler.interest_group ?= {
                save_scope_as = ruling_ig
            }
        }
        else = {
            get_ruler_for:presidential_elective.interest_group ?= {
                save_scope_as = ruling_ig
            }
        }
    }
    ## In a traditional system, the monarch's IG is always the ruling IG until their death
    ## Any way to dethrone this ruling IG involves the death or abdication of this monarch
    ## or switching to an institutional system.
    else_if = {
        limit = {
            modifier:country_government_formation_type_traditional = yes
        }
        if = {
            limit = {
                exists = ruler.interest_group
            }
            ruler.interest_group = {
                save_scope_as = ruling_ig
            }
        }
    }
    ## In the institutional system things are more complicated, the ruling_ig has an bpm_institutional_ruling_ig variable
    ## As long as they have that variable, they are the ruling IG. They can be dethroned by a variety of institutional
    ## actions as well as by more violent means. The ruling IG can also be changed by switching to a popular system.
    else_if = {
        limit = {
            modifier:country_government_formation_type_institutional = yes
        }
        if = {
            limit = {
                var:bpm_headofgov ?= {
                    has_variable = bpm_institutional_ruling_ig
                }
            }
            var:bpm_headofgov ?= {
                save_scope_as = ruling_ig
                set_variable = bpm_institutional_ruling_ig
            }
        }
        # If for some reason no IG is the current ruling IG, and whatever means
        # the previous ruling IG was removed by didn't select a new ruling IG
        # We simply default to the highest clout IG in the country that is not extremist
        else = {
            ordered_interest_group = {
                limit = {
                    bpm_ig_is_very_radical_opposition = no
                }
                alternative_limit = {
                    always = yes
                }
                order_by = ig_clout
                position = 0
                save_scope_as = ruling_ig
            }
        }
    }

    ## Use the selected IG, fallback to ruler
    if = {
        limit = {
            exists = scope:ruling_ig
        }
        scope:ruling_ig = {
            bpm_set_ig_as_hog = yes
        }
    }
    else_if = {
        limit = {
            exists = ruler.interest_group
        }
        ruler.interest_group = {
            bpm_set_ig_as_hog = yes
        }
    }
    else = {
        ordered_interest_group = {
            limit = {
                is_in_government = yes
            }
            alternative_limit = {
                always = yes
            }
            order_by = ig_clout
            position = 0
            bpm_set_ig_as_hog = yes
        }
    }

    # In institutional systems, the ruler should be kept separate from 
    # being the head of government, as the ruler represents something 
    # outside and above the institutional government.
    if = {
        limit = {
            ruler = var:bpm_headofgov.leader
            modifier:country_government_formation_type_institutional = yes
        }
        ordered_scope_character = {
            limit = {
                interest_group ?= prev.var:bpm_headofgov
                is_character_alive = yes
                is_ruler_of_own_country = no
            }
            save_scope_as = selected_char
        }
        if = {
            limit = { NOT = { exists = scope:selected_char } }
            create_character = {
                interest_group = var:bpm_headofgov
                save_scope_as = selected_char
            }
        }
        scope:selected_char = {
            set_as_interest_group_leader = yes
        }
    }
    else = {
        var:bpm_headofgov.leader ?= {
            add_character_role = bpm_character_role_hog
        }
    }
}

bpm_set_ig_as_hog = {
    owner = {
        set_variable = {
            name = bpm_headofgov
            value = prev
        }
    }
}

bpm_make_character_historical_hog = {
    set_variable = bpm_historical_hog
    if = {
        limit = {
            NOT = { has_modifier = bpm_historical_hog }
        }
        add_modifier = {
            name = bpm_historical_hog
            days = normal_modifier_time
        }
    }
    if = {
        limit = { NOT = { exists = interest_group } }
        set_interest_group = owner.var:bpm_headofgov
    }
    if = {
        limit = {
            NOT = { interest_group.leader = this }
        }
        set_as_interest_group_leader = yes
    }
}

bpm_initialize_hog = {
    if = {
        limit = {
            any_scope_character = { has_modifier = bpm_historical_hog }
        }
        random_scope_character = {
            limit = { has_modifier = bpm_historical_hog }
            interest_group = {
                bpm_set_ig_as_hog = yes
                if = {
                    limit = {
                        owner.modifier:country_government_formation_type_institutional = yes
                    }
                    set_variable = bpm_institutional_ruling_ig
                }
            }
        }
    }
    else = {
        ordered_interest_group = {
            limit = {
                is_in_government = yes
            }
            alternative_limit = {
                always = yes
            }
            position = 0
            order_by = ig_clout
            bpm_set_ig_as_hog = yes
        }
    }
    bpm_update_hog = yes
}

bpm_update_regent = {
    if = {
        limit = {
            OR = {
                has_variable = bpm_regency_init
                AND = {
                    bpm_has_gov_regency = yes
                    OR = {
                        NOT = { exists = var:bpm_regent }
                        var:bpm_regent ?= {
                            is_character_alive = no
                        }
                    }
                }
            }
        }
        if = {
            limit = {
                any_scope_character = {
                    interest_group = {
                        is_in_government = yes
                    }
                    bpm_char_can_be_regent = yes
                }
            }
            ordered_interest_group = {
                limit = {
                        is_in_government = yes
                        any_scope_character = {
                            bpm_char_can_be_regent = yes
                        }
                    }
                order_by = ig_clout
                position = 0
                ordered_scope_character = {
                    limit = {
                        bpm_char_can_be_regent = yes
                    }
                    order_by = prominence
                    position = 0
                    bpm_set_character_as_regent = yes
                }
            }
        }
        else = {
            ordered_interest_group = {
                limit = {
                    is_in_government = yes
                }
                order_by = ig_clout
                position = 0
                save_temporary_scope_as = biggest_ig
            }
            create_character = {
	            age = 23
	            interest_group = scope:biggest_ig
	            on_created = {
	            	bpm_set_character_as_regent = yes
	            }
            }
        }
    }
}

bpm_end_regency = {
    bpm_remove_regency_modifier = yes
    every_scope_character = {
        limit = {has_role = bpm_character_role_regent}
        remove_character_role = bpm_character_role_regent
    }
    if = {
        limit = {has_variable = bpm_regent}
        remove_variable = bpm_regent
    }
    ruler ?= {
        set_variable = regency_complete
    }
}

bpm_set_character_as_regent = {
    save_temporary_scope_as = regent_scope
    owner = {
        every_scope_character = {
            limit = {has_role = bpm_character_role_regent}
            remove_character_role = bpm_character_role_regent
        }
        set_variable = {
            name = bpm_regent
            value = prev
        }
        bpm_remove_regency_modifier = yes
        add_regency_modifier = yes
    }
    add_character_role = bpm_character_role_regent
}

bpm_start_regency = {
    set_variable = bpm_regency_init
	bpm_update_regent = yes
	var:bpm_regent = {
		save_scope_as = regent_scope
	}
    remove_variable = bpm_regency_init
}
