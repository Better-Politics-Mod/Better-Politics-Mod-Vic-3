bpm_law_is_always_traditional = {
    OR = {
        type = law_type:law_traditionalism
        type = law_type:law_no_police
        type = law_type:law_no_schools
        type = law_type:law_no_health_system
        type = law_type:law_traditional_ostracism
        type = law_type:law_factory_councils
        type = law_type:law_executive_traditional
    }
}

bpm_law_is_always_constitutional = {
    OR = {
        type = law_type:law_monarchy
        type = law_type:law_presidential_republic
        type = law_type:law_parliamentary_republic
        type = law_type:law_theocracy
        type = law_type:law_council_republic
        type = law_type:law_confederated
        type = law_type:law_federated
        type = law_type:law_devolved
        type = law_type:law_unitary
    }
}

bpm_amendment_is_codification = {
    OR = {
        type = amendment_type:bpm_amendment_constitutional
        type = amendment_type:bpm_amendment_edict
        type = amendment_type:bpm_amendment_traditional
    }
}

bpm_law_is_constitutional = {
    any_scope_amendment = {
        type = amendment_type:bpm_amendment_constitutional
    }
}

bpm_law_is_edict = {
	OR = {
		exists = scope:eval_as_edict
		any_scope_amendment = {
			type = amendment_type:bpm_amendment_edict
		}
	}
}

bpm_law_is_traditional = {
    any_scope_amendment = {
        type = amendment_type:bpm_amendment_traditional
    }
}

bpm_law_is_law = {
    any_scope_amendment = {
        count = 0
        OR = {
            type = amendment_type:bpm_amendment_constitutional
            type = amendment_type:bpm_amendment_edict
            type = amendment_type:bpm_amendment_traditional
        }
    }
}
