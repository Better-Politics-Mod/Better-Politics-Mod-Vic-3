je_bpm_patchwork_governance = {
    dlc = dlc_bpm

    icon = "gfx/interface/icons/event_icons/event_map.dds"

    group = je_group_historical_content

    should_be_pinned_by_default_uninvolved_or_context = yes
	
    is_shown_when_inactive = {
        bpm_country_has_patchwork_governance = yes
    }

    possible = {
        bpm_country_has_patchwork_governance = yes
    }

    map_color = {
		if = {
			limit = { bpm_patchwork_governance_level > 0 }
			lerp = {
                min_color = forest_of_bayonets_ember_orange
                max_color = forest_of_bayonets_fire_red
                factor = {
                    value = bpm_patchwork_governance_level
                    divide = 5
                }
            }
		}
		else = {
			value = "white"
		}
	}

    map_color_second = {
        if = {
            limit = {
                bpm_state_is_dismantling_patchwork_governance = yes
            }
            value = "white"
        }
    }

    on_monthly_pulse = {
        effect = {
            every_scope_state = {
                bpm_patchwork_governance_state_monthly = yes
            }
        }
    }

    outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
            any_scope_state = {
                count = 0
                is_incorporated = yes
                bpm_state_has_patchwork_governance = yes
            }
		}
		effect = {
			trigger_event = {
				id = bpm_patchwork_events.1
                popup = yes
			}
		}

        event_outcome_effect_desc = {
			effect = {
				active_law:lawgroup_centralization = { bpm_amendment_remove_util = { type = bpm_amendment_patchwork_governance } }
			}
		}
	}

    outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			bpm_country_has_patchwork_governance = no
		}
		effect = {
		}
	}
}
